\section{The v2 isoenergetic a-posteriori theorem (I-G1$^{\sharp\sharp}$)}\label{bpI:p2s2}

This part proves the v2 form of Theorem~\ref{bpI:p2s:g1}: the conclusion is the frozen body $\Lean{IG1Body}$ (conclusions (O1)--(O6) of
Theorem~\ref{bpI:p0:g1}), the hypotheses are (I1)--(I13) plus the refined data of \Kref{bp:p2s2:standing}, each implied by (I1)--(I13) with the
v1 values (\Kref{bp:p2s2:defaults}), and the smallness is the disjunction of route A (the isoenergetic step of Part~\ref{bpI:p2s} with the
levers of \Kref{bp:p2s2:constantsA}) and route B (the modified step below). There is no published isoenergetic version of the modified step
(FH24 treat fixed frequencies with first integrals); its isoenergetic ingredients are those of HL19 \S6 (the bordered system, the parallel frequency
correction $\Delta\omega=-\omega\xi^\omega$, the energy row), and the new identities are proved here. For R1
(\texttt{thresholds\_v2.py}): route A $10^{-11.37}$, route B $10^{-12.39}$, v1 $10^{-24.98}$.

\subsection{The bordered modified step}\label{bpI:p2s2:sec-step}

Let $K$ be $(r,\beta,\varepsilon,s)$-admissible (Definition~\ref{bpI:p2:admissible}) with the caps of \Kref{bp:p2s2:step}, $\omega=s\omega^*$, the frozen
bordered torsion $\mathsf T=\bigl(\begin{smallmatrix}T&\omega\\Y&0\end{smallmatrix}\bigr)$, $Y=DH(K)N$, $\abs{\avg{\mathsf T}^{-1}}\le\sigma_T$, the energy error
$E^\omega:=\avg{H\circ K}-E_0$, and the weighted error
\[
\mu:=\max\Bigl\{\frac{\abs{\eta^L}}{\lambda},\ \frac{\abs{\eta^N}}{\gamma\delta^\varkappa},\ \frac{\abs{E^\omega}}{\lambda_\omega}\Bigr\},\qquad X:=\frac\mu{\gamma\delta^{\varkappa+1}},
\]
with weights $\lambda,\lambda_\omega\in(0,1]$ (both $1$ except at stage $0$). Put $\Theta:=\sigma_\omega\abs{\omega^*}\ge\abs\omega$ and $y_0:=M_{h1}\sigma_N\ge\abs Y$ on $\R^n$, where
$M_{h1}\ge\abs{DH}$ and $M_{h2}\ge\abs{D^2H}$ on $D$ (refined data; both $\le M$ by (I6)).

\begin{definition}[Constants of the bordered modified step]\label{bpI:p2s2:constants}
The constants of \Kref{bp:p2s2:constantsB} with the following changes:
\begin{align*}
&C_0:=\sigma_T\max\bigl\{\lambda+t\,\bar c(\delta/r)^\varkappa,\ \lambda_\omega+y_0\,\bar c(\delta/r)^\varkappa\bigr\},\qquad C_{gL}:=\lambda+tC_{\xi N}+\Theta C_0,\qquad
C_{\tilde E}\mathrel{+}=\tilde\sigma_K\Theta C_0,\\
&\bar Q_N\mathrel{+}=(1+n\vartheta_\xi)\Theta C_\Omega\tilde Q_NC_0\vartheta_\xi/C_{\xi L},\qquad
Q_L\mathrel{+}=C_0C_{gL}g_1+C_0\sigma_{N^\top}C_{D\tilde L}\Theta g,\qquad
Q_\omega:=\tfrac{M_{h2}}2C_{\tilde K}^2g_1+n\tilde Q_N\vartheta_\xi(\Theta C_\Omega+g_1),\\
&Q_L\to\max\{Q_L,Q_\omega\},\qquad C_{\Delta\mathsf T}:=\max\bigl\{C_{\Delta T}+\Theta C_0g_1,\ n\Theta C_0g_1+2ng_1\bigr\},\qquad\varrho_\nu:=\Theta C_0\ (\text{weight }\gamma\delta^{\varkappa+1}),
\end{align*}
and $\sigma_TC_{\Delta\mathsf T}X\le\vartheta$ in place of the last step condition. The mirror is \texttt{stageB(\dots, iso=True)} of \texttt{thresholds\_v2.py}.
\end{definition}

\begin{proposition}[The bordered modified step]\label{bpI:p2s2:step}
Under the hypotheses of \Kref{bp:p2s2:step} (with the caps, the bordered torsion and $\mu,X$ as above) and $2^{\varkappa+1}QX\le1$, let
\[
\begin{pmatrix}\xi^N_0\\\xi^\omega\end{pmatrix}:=\avg{\mathsf T}^{-1}\begin{pmatrix}\avg{\eta^L+T\mathcal R\eta^N}\\-E^\omega+\avg{Y\mathcal R\eta^N}\end{pmatrix},\quad
\xi^N:=\xi^N_0-\mathcal R\eta^N,\quad\xi^L:=-\mathcal R(\eta^L-T\xi^N-\omega\xi^\omega),
\]
$\tilde K:=K+N\xi^N$, $\bar K:=\tilde K\circ(\mathrm{id}+\xi^L)$, $\bar\omega:=(1-\xi^\omega)\omega=\bar s\omega^*$. Then $\abs{\bar s-s}\abs{\omega^*}\le\Theta C_0\mu$, the conclusions of
\Kref{bp:p2s2:step} hold for $(\bar K,\bar\omega)$ with the constants of Definition~\ref{bpI:p2s2:constants}, and $\abs{\bar E^\omega}\le Q_\omega\mu^2/(\gamma\delta^{\varkappa+1})$.
\source{HL19 Lemma~6.1 and Lemma~6.2, pp.~46--51; FH24 Lemma~3.6; Lemma~\ref{bpI:p1:border-solve}.}
\sourcegap{No source treats the isoenergetic modified step; Steps~\ref{stepI:p2s2:st-w}--\ref{stepI:p2s2:st-en} supply the identities.}
\end{proposition}

\begin{proof}
\proofstep{The bordered averages}\label{stepI:p2s2:st-av}
The first row of $\avg{\mathsf T}(\xi^N_0,\xi^\omega)$ gives $\avg{\eta^L-T\xi^N-\omega\xi^\omega}=0$, so $\xi^L$ is defined; the second gives $\avg{Y\xi^N}=-E^\omega$.
By \Kref{bp:p2s2sd:avg} the right side is bounded by $\max\{\lambda+t\bar c(\delta/r)^\varkappa,\lambda_\omega+y_0\bar c(\delta/r)^\varkappa\}\mu$ in the bordered
row-sum norm, so $\abs{\xi^N_0},\abs{\xi^\omega}\le C_0\mu$ (Lemma~\ref{bpI:p1:border-solve}(c) structure) and $\abs{\Delta\omega}=\abs\omega\abs{\xi^\omega}\le\Theta C_0\mu$.

\proofstep{The frequency in the error}\label{stepI:p2s2:st-w}
$\tilde E_{\bar\omega}:=X_H(\tilde K)-D\tilde K\bar\omega=\tilde E_\omega+\xi^\omega D\tilde K\omega$, and $\bar E=\tilde E_{\bar\omega}\circ\varphi-(\tilde L\circ\varphi)\partial_{\bar\omega}\xi^L$ with
$\partial_{\bar\omega}\xi^L=(1-\xi^\omega)\partial_\omega\xi^L$. Hence $\abs{\tilde E_{\bar\omega}}\le(C_{\tilde E}+\tilde\sigma_K\Theta C_0)\mu$, and $\tilde L^\top\Omega_0\tilde E_{\bar\omega}=\tilde\eta^N+\xi^\omega\Omega_{\tilde K}\omega$,
so $\bar\eta^N=(I+D\xi^L)^\top(\tilde\eta^N\circ\varphi+\xi^\omega\Omega_{\tilde K}\circ\varphi\,\omega-\Omega_{\tilde K}\circ\varphi\,\partial_{\bar\omega}\xi^L)$; the middle term is cubic and gives
the increment of $\bar Q_N$ (with $C_{\xi L}X\le\vartheta_\xi$).

\proofstep{The tangential error}\label{stepI:p2s2:st-l}
$-N^\top\Omega_0\tilde E_{\bar\omega}=\eta^L-T\xi^N-\omega\xi^\omega-N^\top\Omega_0\Delta^2X-\Omega_N\eta^N-\xi^\omega N^\top\Omega_0D(N\xi^N)\omega$ (as $N^\top\Omega_0L=I$), and
$N^\top\Omega_0L\partial_{\bar\omega}\xi^L=-(1-\xi^\omega)(\eta^L-T\xi^N-\omega\xi^\omega)$. The difference leaves $\xi^\omega(\eta^L-T\xi^N-\omega\xi^\omega)$, of norm $\le C_0C_{gL}\mu^2$,
and $\xi^\omega N^\top\Omega_0D(N\xi^N)\omega$, of norm $\le C_0\sigma_{N^\top}C_{D\tilde L}\Theta\mu^2/\delta$; the other terms are those of \Kref{step:p2s2:st-etal}.

\proofstep{The energy error}\label{stepI:p2s2:st-en}
$\avg{H\circ\tilde K}=\avg{H\circ K}+\avg{Y\xi^N}+\avg{\Delta^2H}=E_0+\avg{\Delta^2H}$ by the second bordered row, with $\abs{\Delta^2H}\le\frac{M_{h2}}2\abs{N\xi^N}^2$. For the
composition, $D(H\circ\tilde K)=DH(\tilde K)\tilde L=-X_H(\tilde K)^\top\Omega_0\tilde L=-\omega^\top\Omega_{\tilde K}+(\tilde\eta^N)^\top$, so
$\abs{\avg{H\circ\tilde K\circ\varphi-H\circ\tilde K}}\le n(\Theta\abs{\Omega_{\tilde K}}+\abs{\tilde\eta^N})\abs{\xi^L}$, which is cubic; with $C_{\xi L}X\le\vartheta_\xi$ this is the bound $Q_\omega$.

\proofstep{The bordered torsion}\label{stepI:p2s2:st-t}
$\avg{\bar{\mathsf T}}-\avg{\mathsf T}$ has blocks $\avg{\bar T-T}$ (\Kref{step:p2s2:st-fr}; the torsion polynomial involves $\bar\omega$ only through $D\bar E_{\bar\omega}$),
$\bar\omega-\omega$, and $\avg{\bar Y-Y}$ with $Y=\omega^\top-(\eta^L)^\top$ (Lemma~\ref{bpI:p1:last-row}(a) and its algebraic content for the new pair, as in
Step~\ref{stepI:p2s:tw-y}), so $\abs{\bar Y-Y}\le n\abs{\Delta\omega}+n(\abs{\eta^L}+\abs{\bar\eta^L})\le(n\Theta C_0+2n)g_1X$; \Kref{bp:p2:matrix}(c) gives $C_{\Delta\mathsf T}$, and
\Kref{bp:p2s:neumann} the inverse.
\end{proof}

\subsection{The theorem}\label{bpI:p2s2:sec-g1}

\begin{theorem}[I-G1 with v2 constants]\label{bpI:p2s2:g1}
Let the data of Theorem~\ref{bpI:p0:g1} and the refined data of \Kref{bp:p2s2:standing} be given, and put
$\lambda_0:=\sigma_{N^\top}\gamma\delta_0^\varkappa/\sigma_{L^\top}$, $\lambda_{\omega,0}:=\gamma\delta_0^\varkappa/\sigma_{L^\top}$ and $X_0:=\sigma_{L^\top}e/(\gamma^2\delta_0^{2\varkappa+1})$. If the conditions
\Kref{bp:p2s2:J1}--\Kref{bp:p2s2:J3} of \Kref{bp:p2s2:engine} hold for the bordered modified step (caps of \Kref{bp:p2s2:g1} and the frequency cap with budget $\nu/2$
and rate $\varrho_\nu$), or route A of \Kref{bp:p2s2:g1} holds for the isoenergetic step of Part~\ref{bpI:p2s} with the levers, then (O1)--(O6) hold, with
$\abs{K_\infty-K}$ bounded by the ball sum and $\abs{s_\infty\omega^*-s\omega^*}$ by the frequency sum of (J3).
\source{HL19 Theorem~3.6 and Remark~3.7; Theorem~\ref{bpI:p2s:g1}; FH24 Theorem~2.18.}
\end{theorem}

\begin{proof}
\proofstep{The engine}\label{stepI:p2s2:g1}
At stage $0$, $\abs{\eta^L}\le\sigma_{N^\top}e$, $\abs{\eta^N}\le\sigma_{L^\top}e$, $\abs{E^\omega}\le e$, so $X\le X_0$ with the weights $\lambda_0,\lambda_{\omega,0}$. The frequency cap gives
$\nu(s_j)\ge\nu/2$ for every $j$ as in Step~\ref{stepI:p2s:sc-s}; Proposition~\ref{bpI:p2s2:step} is the abstract step and \Kref{bp:p2s2:engine} the iteration and limit.
(O2) is Step~\ref{stepI:p2:lim-inv}; (O3) is Lemma~\ref{bpI:p2:energy-limit} with $\abs{E^\omega_j}\le\lambda_\omega\mu_j\to0$ and Lemma~\ref{bpI:p1:energy-conservation}(b);
(O4)--(O6) as Step~\ref{stepI:p2s:g1-o}.
\end{proof}

\begin{corollary}[Frozen form and hybrid]\label{bpI:p2s2:frozen}
\Kref{bp:p2s2:frozen-form} and \Kref{bp:p2s2:hybrid} hold verbatim for the isoenergetic theorem, with $C_2^{\sharp\sharp},C_3^{\sharp\sharp}$ the ball and frequency sums of
(J3) at $X_\star$ times $\gamma^2\rho^{2\tau}/e$. In the SO checker only \Lean{checkSmallness} changes: it tests $X_0\le X_\star$ for a certified $X_\star$ (or the
two routes' finite lists of inequalities directly); every other sub-check proves hypotheses of \Lean{IG1Body}, unchanged.
\source{Ours.}
\end{corollary}

\begin{proof}
\proofstep{As at fixed frequency}\label{stepI:p2s2:ff}
The proofs of \Kref{bp:p2s2:frozen-form}, \Kref{bp:p2s2:hybrid} use only monotonicity in $X_0$ and one application of the step.
\end{proof}

\begin{remark}[Deviations]\label{bpI:p2s2:dev}
(W1$^{\sharp\sharp}$) the energy error carries the weight $\lambda_\omega$ at stage $0$ and $1$ afterwards (HL19 use $\norm{E_c}=\max\{\norm E,\abs{E^\omega}\}$);
(W2$^{\sharp\sharp}$) the averages of the bordered system use the full bite (\Kref{bp:p2s2sd:avg}; HL19 (6.29) use the bite $\delta$);
(W3$^{\sharp\sharp}$) $\Theta=\sigma_\omega\abs{\omega^*}$ (HL19 (6.30), $C_\omega$) instead of $2n\sigma_K\sigma_B(M+1)$. The errata of Remarks~\Kref{bp:p2s2:register} and~\ref{bpI:p2s:register}
stand.
\end{remark}
