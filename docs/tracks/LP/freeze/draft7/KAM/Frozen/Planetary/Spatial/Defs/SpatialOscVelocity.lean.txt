/-- The actual heliocentric velocity `V_i = p_i/ν_i + μ ∑_j p_j` (`LP:MAIN (10)`, `L16`, `d = 3`).

def spatialOscVelocity {n : ℕ} (ν : Fin n → ℝ) (μ : ℝ) (z : Phase (n * 3))
    (i : Fin n) : Fin 3 → ℝ :=
