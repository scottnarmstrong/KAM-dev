/-- The feasible set `{(u,v) ∈ [0,1]² : a_i(1+u) ≥ a_j(1-v)}` of the pair `(i,j)`. -/
def pairFeasible {n : ℕ} (ν I : Fin n → ℝ) (i j : Fin n) : Set (ℝ × ℝ) :=
  {uv | uv.1 ∈ Set.Icc (0 : ℝ) 1 ∧ uv.2 ∈ Set.Icc (0 : ℝ) 1 ∧
    actionAxis ν I j * (1 - uv.2) ≤ actionAxis ν I i * (1 + uv.1)}
