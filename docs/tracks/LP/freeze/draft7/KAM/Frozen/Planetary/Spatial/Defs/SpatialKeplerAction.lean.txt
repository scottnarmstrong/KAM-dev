/-- Kepler action `Λ_i = ν_i √a_i`. -/
def spatialKeplerAction {n : ℕ} (ν : Fin n → ℝ) (z : Phase (n * 3)) (i : Fin n) : ℝ :=
  ν i * Real.sqrt (spatialKeplerAxis ν z i)
