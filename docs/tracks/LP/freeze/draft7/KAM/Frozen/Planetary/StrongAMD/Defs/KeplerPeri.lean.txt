/-- Canonical perihelion `a_i (1 - e_i)` (`a_i = keplerAxis`, `e_i = |keplerEccVector|`). -/
def keplerPeri {n : ℕ} (ν : Fin n → ℝ) (z : Phase (n * 2)) (i : Fin n) : ℝ :=
  keplerAxis ν z i * (1 - planeLength (keplerEccVector ν z i))
