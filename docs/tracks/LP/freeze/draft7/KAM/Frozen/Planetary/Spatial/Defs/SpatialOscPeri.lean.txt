/-- Osculating perihelion `a^osc (1 - e^osc)`. -/
def spatialOscPeri {n : ℕ} (ν : Fin n → ℝ) (μ : ℝ) (z : Phase (n * 3)) (i : Fin n) : ℝ :=
  spatialOscAxis ν μ z i * (1 - spatialOscEcc ν μ z i)
