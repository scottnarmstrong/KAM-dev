open Lean Meta Elab Command

/-- The constants reachable from `n` through definition bodies (theorems, i.e. proofs, are not
followed). -/
partial def reach (env : Environment) (start : Name) : NameSet := Id.run do
  let mut seen : NameSet := {}
  let mut todo : List Name := [start]
  while !todo.isEmpty do
    let n := todo.head!
    todo := todo.tail!
    if seen.contains n then continue
    seen := seen.insert n
    match env.find? n with
    | some (.defnInfo d) => todo := d.value.getUsedConstants.toList ++ d.type.getUsedConstants.toList ++ todo
    | some (.opaqueInfo d) => todo := d.value.getUsedConstants.toList ++ todo
    | some (.inductInfo i) => todo := i.ctors ++ todo
    | some (.ctorInfo c) => todo := c.type.getUsedConstants.toList ++ todo
    | _ => pure ()
  return seen

elab "#reaches " a:ident b:ident : command => do
  let env ← getEnv
  let s := reach env a.getId
  logInfo m!"{a.getId} reaches {b.getId}: {s.contains b.getId} ({s.size} constants)"
