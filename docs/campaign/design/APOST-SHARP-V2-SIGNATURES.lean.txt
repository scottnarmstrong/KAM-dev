/-- `δ₀ = min {1, (ρ - ρ∞)(a - 1)/(3a)}`. -/
abbrev bite0R (a ρ ρinf : ℝ) : ℝ := min 1 ((ρ - ρinf) * (a - 1) / (3 * a))
/-- `δ_s = δ₀ a^{-s}`. -/
abbrev biteSR (a ρ ρinf : ℝ) (s : ℕ) : ℝ := bite0R a ρ ρinf / a ^ s
/-- `r_s = ρ - 3δ₀(1 - a^{-s})/(1 - a^{-1})`, so `r_s - 3δ_s = r_{s+1}`. -/
abbrev stripSR (a ρ ρinf : ℝ) (s : ℕ) : ℝ := ρ - 3 * bite0R a ρ ρinf * (1 - a⁻¹ ^ s) / (1 - a⁻¹)

/-- `X_{s+1} = a^p Q_s X_s²`. -/
def xSeqR (a p : ℝ) (Q : ℕ → ℝ) (X0 : ℝ) : ℕ → ℝ
  | 0 => X0
  | s + 1 => a ^ p * Q s * xSeqR a p Q X0 s ^ 2

/-- The ball sum with the tail factor `1/(1 - ȳ)`. -/
def stageBallR {k : ℕ} (a κ γ ρ ρinf μ₀ ybar : ℝ) (C : ℝ → StageConsts k) (kpre : ℕ) (Ctail : StageConsts k) : ℝ :=
  (∑ s ∈ Finset.range kpre, (C (biteSR a ρ ρinf s)).CDK * biteSR a ρ ρinf s *
      xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) s) +
    Ctail.CDK * biteSR a ρ ρinf kpre *
      xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) kpre / (1 - ybar)

/-- Blueprint `bp:p2s2:engine` with ratio `a > 1` and decay bound `ȳ ∈ (0,1)` (FH24 §3.3: `a`, `κ`); for `a = 2`, `ȳ = 1/2`
it is `StageScheme.converge_checked` (B5). Route B: exponent `κ + 1`; route A uses B9's `SchemeA`. -/
theorem StageScheme.converge_ratio {n k : ℕ} {α : Type*} {S : StageScheme n k α} {κ γ ρ ρinf β₀ μ₀ a ybar : ℝ}
    {C : ℝ → StageConsts k} {σ m₀ : Fin k → ℝ} {x₀ : α} (kpre : ℕ) (Ctail : StageConsts k)
    (hS : S.IsNewton κ γ (bite0R a ρ ρinf) C σ) (ha : 1 < a) (hy0 : 0 < ybar) (hy1 : ybar < 1) (hγ : 0 < γ)
    (hρinf : 0 < ρinf) (hρ : ρinf < ρ) (hx₀ : S.Adm ρ β₀ μ₀ m₀ x₀)
    (htail : ∀ δ, 0 < δ → δ ≤ biteSR a ρ ρinf kpre →
      (C δ).Q ≤ Ctail.Q ∧ (∀ X, Ctail.cond X → (C δ).cond X) ∧ (C δ).CDK ≤ Ctail.CDK ∧
        (C δ).Cp ≤ Ctail.Cp ∧ ∀ i, (C δ).rate i * (C δ).wt i ≤ Ctail.rate i * Ctail.wt i)
    (hJ1 : ∀ s < kpre, (C (biteSR a ρ ρinf s)).cond
      (xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) s))
    (hJ2 : a ^ (κ + 1) * Ctail.Q *
      xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) kpre ≤ ybar)
    (hJ3 : ∀ i, m₀ i + (∑ s ∈ Finset.range kpre, (C (biteSR a ρ ρinf s)).rate i * (C (biteSR a ρ ρinf s)).wt i *
        xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) s) +
      Ctail.rate i * Ctail.wt i *
        xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q) (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) kpre /
          (1 - ybar) ≤ σ i)
    (hnn : ∀ δ, 0 < δ → δ ≤ bite0R a ρ ρinf →
      0 ≤ (C δ).Q ∧ 0 ≤ (C δ).CDK ∧ ∀ i, 0 ≤ (C δ).rate i * (C δ).wt i)
    (hcondT : ∀ X, 0 ≤ X → X ≤ xSeqR a (κ + 1) (fun s => (C (biteSR a ρ ρinf s)).Q)
      (μ₀ / (γ * bite0R a ρ ρinf ^ (κ + 1))) kpre → Ctail.cond X)
    (hball : stageBallR a κ γ ρ ρinf μ₀ ybar C kpre Ctail < β₀) :
    ∃ (Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (pinf : ℝ),
      DifferentiableOn ℂ Kinf (complexStrip n ρinf) ∧ IsPrimaryOnStrip Kinf ρinf ∧ IsRealOnReals Kinf ∧
      (∀ s, ∃ r β μ m, S.Adm r β μ m (Nat.rec x₀ (fun j x => S.step (stripSR a ρ ρinf j) (biteSR a ρ ρinf j) x) s) ∧
        ρinf < r ∧ (∀ i, m i ≤ σ i)) ∧
      TendstoUniformlyOn (fun s => S.tor (Nat.rec x₀ (fun j x => S.step (stripSR a ρ ρinf j) (biteSR a ρ ρinf j) x) s))
        Kinf atTop (complexStrip n ρinf) ∧
      (∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - S.tor x₀ θ‖ ≤ stageBallR a κ γ ρ ρinf μ₀ ybar C kpre Ctail) := sorry

end KAM.Sharp2

variable {n : ℕ}

/-- The commutator `A Ω₀ - Ω₀ A`, `A = DX_h(z)` (refined data (R1): `c_Th`, `c_DTh`). -/
def hamComm (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (z : (Fin n → ℂ) × (Fin n → ℂ)) :
    Matrix (Fin n ⊕ Fin n) (Fin n ⊕ Fin n) ℂ :=
  sqMat (fderiv ℂ (hamiltonianVectorFieldℂ h) z) * omegaZeroMatrix n -
    omegaZeroMatrix n * sqMat (fderiv ℂ (hamiltonianVectorFieldℂ h) z)

/-- The frame part `𝒯_ab = B Lᵀ (A Ω₀ - Ω₀ A) L B` of the torsion polynomial (`bp:p2s:torsion-identity`). -/
def torsionAB (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (θ : Fin n → ℂ) : Matrix (Fin n) (Fin n) ℂ :=
  (gramMatrix K θ)⁻¹ * (tangentMat K θ)ᵀ * hamComm h (K θ) * tangentMat K θ * (gramMatrix K θ)⁻¹

/-- `⟨𝒯_ab⟩` over the real torus (as `torsionAverage`). -/
def torsionABAverage (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) :
    Matrix (Fin n) (Fin n) ℂ :=
  fun i j => ∫ θ in Set.pi Set.univ (fun _ : Fin n => Set.Icc (0 : ℝ) 1), torsionAB h K (fun l => (θ l : ℂ)) i j

/-- `bp:p2s2:torsion-local`, the frame part: `|𝒯_ab| ≤ c_Th |Nᵀ| |N|`. -/
theorem norm_torsionAB_le {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)}
    {θ : Fin n → ℂ} {sN sNt cTh : ℝ} (hG : IsUnit (gramMatrix K θ).det) (hN : ‖normalMat K θ‖ ≤ sN)
    (hNt : ‖(normalMat K θ)ᵀ‖ ≤ sNt) (hc : ‖hamComm h (K θ)‖ ≤ cTh) :
    ‖torsionAB h K θ‖ ≤ sNt * cTh * sN := sorry

/-- `bp:p2s2:torsion-local`, the local part: `|T - 𝒯_ab| = |𝒯_F + 𝒯_c| ≤ σ_{Nᵀ} σ_B |DE| + σ_B³ w |Ω_K|`. -/
theorem norm_torsion_sub_torsionAB_le {τ M σK σB σT γ : ℝ} {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {ω : Fin n → ℝ} (hS : Standing n τ M σK σB σT 𝔅 h ω γ) {r : ℝ}
    (hr : 0 < r) {K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} (hK : DifferentiableOn ℂ K (complexStrip n r))
    (hK𝔅 : ∀ θ ∈ complexStrip n r, K θ ∈ 𝔅) (hG : ∀ θ ∈ complexStrip n r, IsUnit (gramMatrix K θ).det)
    {θ : Fin n → ℂ} (hθ : θ ∈ complexStrip n r) {sB sNt w dE oK : ℝ}
    (hB : ‖(gramMatrix K θ)⁻¹‖ ≤ sB) (hNt : ‖(normalMat K θ)ᵀ‖ ≤ sNt) (hDE : ‖errDerivMat h K ω θ‖ ≤ dE)
    (hW : ‖(tangentMat K θ)ᵀ * ((linMat h K θ)ᵀ + linMat h K θ) * tangentMat K θ -
      (errDerivMat h K ω θ)ᵀ * tangentMat K θ - (tangentMat K θ)ᵀ * errDerivMat h K ω θ‖ ≤ w)
    (hΩ : ‖lagrangeMat K θ‖ ≤ oK) :
    ‖torsion h K ω θ - torsionAB h K θ‖ ≤ sNt * sB * dE + sB ^ 3 * w * oK := sorry

/-- `bp:p2s2:torsion-ref`: Neumann against the initial average. -/
theorem inv_le_of_ref {A A0 : Matrix (Fin n) (Fin n) ℂ} {m σ : ℝ} (hA0 : IsUnit A0.det) (hm : ‖A0⁻¹‖ ≤ m)
    (hm0 : 0 < m) (hmσ : m < σ) (hd : ‖A - A0‖ ≤ (σ - m) / (σ * m)) : IsUnit A.det ∧ ‖A⁻¹‖ ≤ σ := sorry

/-- The pullback of the Lagrangian defect (option L10, `step:p2s2:selfcons`): `Ω_{K̃∘φ} = Dφᵀ (Ω_{K̃}∘φ) Dφ`. -/
theorem norm_lagrangeMat_comp_le {Kt : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ξ : (Fin n → ℂ) → (Fin n → ℂ)}
    {θ : Fin n → ℂ} {b oK : ℝ} (hKt : DifferentiableAt ℂ Kt (θ + ξ θ)) (hξ : DifferentiableAt ℂ ξ θ)
    (hb : ∀ i, ∑ j, ‖fderiv ℂ ξ θ (Pi.single j 1) i‖ ≤ b) (hbT : ∀ j, ∑ i, ‖fderiv ℂ ξ θ (Pi.single j 1) i‖ ≤ b)
    (hΩ : ‖lagrangeMat Kt (θ + ξ θ)‖ ≤ oK) :
    ‖lagrangeMat (fun θ' => Kt (θ' + ξ θ')) θ‖ ≤ (1 + b) ^ 2 * oK := sorry

/-- The inputs of one repaired modified step (`bp:p2s2:constantsB`): caps, Hamiltonian data `M₁ M₂ c_Th c_DTh`, the
torsion slack `ts` (so `t = c_Th σ_{Nᵀ} σ_N + ts`), the local torsion budget `bF`, the small-divisor constants `c`
(bite `δ`), `c1` (Rüssmann-Cauchy, from the split bite), `cavg` (average constant), the ratios, the schedule ratio `a`,
the weight `λ`, `ḡ = γ̄`, the bite value `δ` used in the constants (the actual bite may be any `δ' ≤ δ`), and the options
L10 (`selfcons`), stage-0 (`stage0`, `ePer = e/μ₀`) and iso. -/

  n : ℕ
  κ : ℝ
  M1 : ℝ
  M2 : ℝ
  cTh : ℝ
  cDTh : ℝ

  sB : ℝ

  ts : ℝ

  lam : ℝ

  δ : ℝ

  stage0 : Bool := false
  ePer : ℝ := 0
  iso : Bool := false
  Θ : ℝ := 0
  y0 : ℝ := 0
  Mh2 : ℝ := 0
  lamW : ℝ := 1

  CΩ : ℝ

  CDLtT : ℝ

/-- Blueprint `bp:p2s2:constantsB` and `bpI:p2s2:constants` (repaired: torsion against the initial torus, iso factor
`a_ω = 1 + θ_ω`, polynomial `Q̄_N`, option L10, schedule ratio `a`). -/
def stepConstsB (p : StepInB) : StepConstsB :=
  let n : ℝ := p.n
  let g := p.γb * p.δ ^ p.κ
  let g1 := g * p.δ
  let sKt := (1 + p.θ) * p.sK
  let t := p.cTh * p.sNt * p.sN + p.ts
  let aw : ℝ := if p.iso then 1 + p.θω else 1
  let CΩ := 2 * (n - 1) * p.c1
  let CEL := (p.sK + p.sN * p.q) / (1 - p.q ^ 2)
  let CEN := (p.sN + p.sK * p.q) / (1 - p.q ^ 2)
  let CE0 := p.lam * CEL + g * CEN
  let CE := if p.stage0 then min CE0 p.ePer else CE0
  let CDE0 := n * (p.sK + p.sN) * (max p.lam g + (p.sLt + p.sNt) * CE) / (1 - p.q)
  let CDE := if p.stage0 then min CDE0 (n * p.ePer) else CDE0
  let CNN := n * (p.lam + p.sNt * CE)
  let C0 := if p.iso then p.sT * max (p.lam + t * p.cavg) (p.lamW + p.y0 * p.cavg) else p.sT * (p.lam + t * p.cavg)
  let CxiN := C0 + p.c
  let CKt := p.sN * CxiN
  let CDLt := n * p.sN * (C0 + p.c1)
  let CDLtT := 2 * n * p.sN * (C0 + p.c1)
  let CdN := (p.M1 * p.sK + n) * p.sB + p.sN * ((p.sLt * p.M1 + 2 * n) * p.sK + p.sLt * (p.M1 * p.sK + n)) * p.sB
  let CXN := p.M1 * p.sN + CdN
  let CEt := CE + CXN * CxiN + p.sN * g + p.M2 / 2 * CKt + (if p.iso then sKt * p.Θ * C0 else 0)
  let QtN := CDLtT * CEt + CNN * CxiN + p.δ * p.sLt * (p.M2 / 2) * CKt ^ 2
  let CgL := p.lam + t * CxiN + (if p.iso then p.Θ * C0 else 0)
  let CxiL := p.c * CgL
  let CDK := sKt * CxiL + g * CKt
  let CEb := CEt + aw * sKt * CgL
  let CG := n * p.sLt + 2 * n * p.sK
  let CBb := p.sB ^ 2 * CG * CDK / (1 - p.θ)
  let CNb := n * CDK * p.sB + p.sK * CBb
  let CNbT := CBb * p.sLt + 2 * n * CDK * p.sB
  let w := 2 * p.M1 * p.sLt * p.sK + 2 * n * p.sK + n * p.sLt
  let Cab0 := p.cTh * (CNbT * p.sN + p.sNt * CNb) + p.cDTh * p.sNt * p.sN * CDK * p.δ
  let Cab := if p.iso then max (Cab0 + p.Θ * C0 * g1) (n * p.Θ * C0 * g1) else Cab0
  let Bsup := (p.sNt * p.sB * CDE + p.sB ^ 3 * w * CΩ) * g
  let Bloc := if p.iso then max Bsup (n * g1) else Bsup
  let QbN := if p.iso then (1 + n * p.θξ) * (1 + (1 + aw) * p.θΩ) * QtN else (1 + n * p.θξ) * (1 + p.θΩ) * QtN
  let QL1 := p.sNt * (p.M2 / 2) * CKt ^ 2 * g1 + p.sB ^ 2 * CΩ * g ^ 2 + n * p.sNt * CEt * CxiL +
    aw * p.sNt * CgL * (n * sKt * CxiL + g * CDLt) +
    (if p.iso then C0 * CgL * g1 + C0 * p.sNt * CDLt * p.Θ * g else 0)
  let QLc := QL1 + CNbT * CEb
  let QLs := if p.selfcons then min QLc ((QL1 + p.θN * (CEN / CEL) * QbN * g) / (1 - p.θN)) else QLc
  let Qw := p.Mh2 / 2 * CKt ^ 2 * g1 + n * QtN * p.θξ * (p.Θ * CΩ + g1)
  let QL := if p.iso then max QLs Qw else QLs
  ⟨CΩ, CEL, CEN, CE, CDE, C0, CxiN, CKt, CDLt, CDLtT, CEt, QtN, CgL, CxiL, CDK, CEb, CG, CBb, CNb, CNbT, Cab, Bsup,
    Bloc, QbN, QL, max QL (p.a ^ p.κ * QbN), if p.iso then p.Θ * C0 else 0⟩

/-- The step conditions (`bp:p2s2:constantsB`); every one is `coef · X^p ≤ b` with `coef ≥ 0`. No Neumann condition
for `⟨T̄⟩` and no local contraction premise (`bp:p2s2:torsion-ref`). -/
def stepCondB (p : StepInB) (X : ℝ) : Prop :=
  let k := stepConstsB p
  let g := p.γb * p.δ ^ p.κ
  max 1 (p.sB ^ 2) * k.CΩ * g * X ≤ p.q ∧ k.CDE * g * X ≤ p.n ∧ k.CxiL * X ≤ p.θξ ∧
    k.CΩ * k.CgL * X ≤ p.θΩ ∧ k.CKt * (g * p.δ) * X ≤ 1 ∧ k.CDLt * g * X ≤ p.θ * p.sK ∧
    k.CDLtT * g * X ≤ p.θ * p.sLt ∧ p.sB * k.CG * k.CDK * X ≤ p.θ ∧ k.Bsup * X ≤ p.ts ∧ k.Bloc * X ≤ p.bF ∧
    (p.iso = true → k.C0 * (g * p.δ) * X ≤ p.θω) ∧
    (p.selfcons = true → k.CNbT * k.CEL * X ≤ p.θN ∧
      max 1 (p.sB ^ 2) * (1 + p.n * p.θξ) ^ 2 * k.CΩ * k.QtN * g * X ^ 2 ≤ p.q)

/-- Admissible inputs (`bp:p2s2:data-ok`, audit F1). -/
structure StepInB.Ok (p : StepInB) : Prop where
  one_le_n : 1 ≤ p.n
  one_le_κ : 1 ≤ p.κ

/-- Nonnegativity of every constant on admissible inputs (the engine's `hnn`). -/

/-- `ξᴺ₀ = ⟨T⟩⁻¹ ⟨ηᴸ + T 𝓡ηᴺ⟩`. -/
def modXiN0 (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) : Fin n → ℂ :=
  (torsionAverage h K ω)⁻¹ *ᵥ avg (fun θ => etaL h K ω θ + torsion h K ω θ *ᵥ cohR ω (etaN h K ω) θ)

/-- `ξᴺ = ξᴺ₀ - 𝓡ηᴺ`. -/
def modXiN (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : Fin n → ℂ :=
  modXiN0 h K ω - cohR ω (etaN h K ω) θ

/-- `K̃ = K + N ξᴺ`. -/
def modKtilde (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : (Fin n → ℂ) × (Fin n → ℂ) :=
  K θ + (sumVec n).symm (normalMat K θ *ᵥ modXiN h K ω θ)

/-- `ξᴸ = -𝓡(ηᴸ - T ξᴺ)`. -/
def modXiL (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : Fin n → ℂ :=
  -cohR ω (fun θ' => etaL h K ω θ' - torsion h K ω θ' *ᵥ modXiN h K ω θ') θ

/-- `K̄ = K̃ ∘ (id + ξᴸ)` on `T_w`, `K` elsewhere (`w = r - 2δ`). -/
def modNewton (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (w : ℝ) : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ) :=
  fun θ => if θ ∈ complexStrip n w then modKtilde h K ω (θ + modXiL h K ω θ) else K θ

  normal : ∀ θ ∈ complexStrip n ρ, ‖normalMat K θ‖ ≤ I.iN
  normalT : ∀ θ ∈ complexStrip n ρ, ‖(normalMat K θ)ᵀ‖ ≤ I.iNt

/-- The stage-0 weights fit every body scale `γ ≤ γ̄` (clipped weights, audit iso F4): with
`p̄ = max {σ_{Nᵀ}, σ_{Lᵀ}/(γ̄ δ₀^τ), 1 (iso)}`, `λ₀ ≥ σ_{Nᵀ}/p̄`, `λ_{ω,0} ≥ 1/p̄`, `e/μ₀ ≤ 1/p̄`. -/
structure NumB.Fits {n : ℕ} (N : NumB n) (I : InitB) : Prop where
  lam0 : I.iNt / max (max I.iNt (I.iLt / (N.γb * N.δ0 ^ N.τ))) (if N.iso then 1 else 0) ≤ N.lam0
  lamW0 : N.iso = true → 1 / max (max I.iNt (I.iLt / (N.γb * N.δ0 ^ N.τ))) 1 ≤ N.lamW0
  ePer0 : 1 / max (max I.iNt (I.iLt / (N.γb * N.δ0 ^ N.τ))) (if N.iso then 1 else 0) ≤ N.ePer0

/-- The initial cap values. -/
def initB (I : InitB) (σK σB μ : ℝ) : Fin 7 → ℝ := ![σK - μ, I.iLt, σB - μ, I.iN, I.iNt, 0, 0]

/-- The caps: frames as in `N`, `Tab` with the Neumann budget `β_T - 2 b_F` (`bp:p2s2:torsion-ref`,
`β_T = (σ̂_T - m_T)/(σ̂_T m_T)`, `m_T = σ_T - μ`), the frequency with budget `ν/2`. -/
def capB {n : ℕ} (N : NumB n) (σT μ ν : ℝ) : Fin 7 → ℝ :=
  ![N.sK, N.sLt, N.sB, N.sN, N.sNt, (N.sT - (σT - μ)) / (N.sT * (σT - μ)) - 2 * N.bF, ν / 2]

/-- (J1)–(J4) of `bp:p2s2:engine` for route B at `X⋆` (all caps in J3, tail conditions on `[0, X_kp]`). -/
structure routeBOk {n : ℕ} (N : NumB n) (σ m₀ : Fin 7 → ℝ) (β Xs : ℝ) : Prop where
  Xs_nonneg : 0 ≤ Xs
  J1 : ∀ s < N.kp, stepCondB (N.inp s) (N.xS Xs s)
  J1t : ∀ X, 0 ≤ X → X ≤ N.xS Xs N.kp → stepCondB (N.inp N.kp) X
  J2 : N.a ^ (N.τ + 1) * N.Qs N.kp * N.xS Xs N.kp ≤ N.ybar
  J3 : ∀ i, m₀ i + (∑ s ∈ Finset.range N.kp, N.rate s i * N.wt s i * N.xS Xs s) +
    N.rate N.kp i * N.wt N.kp i * N.xS Xs N.kp / (1 - N.ybar) ≤ σ i
  J4 : N.ballSum Xs < β

/-- The coefficient `p̄₀` of the stage-0 error in `X₀` (fixed: `σ_{Nᵀ}`, iso: `max {σ_{Nᵀ}, 1}`). -/
def pN {n : ℕ} (N : NumB n) (I : InitB) : ℝ := if N.iso then max I.iNt 1 else I.iNt

/-- `C₁` of route B (`bp:p2s2:frozen-form`). -/
def c1B {n : ℕ} (N : NumB n) (I : InitB) (Xs : ℝ) : ℝ :=
  N.ρ ^ (4 * N.τ) * max (I.iLt * N.γb ^ 2) (pN N I * N.γb ^ 3 * N.δ0 ^ N.τ) / (Xs * N.δ0 ^ (2 * N.τ + 1))

/-- `C₂` of route B: the ball sum, without `γ̄²` (audit F6). -/
def c2B {n : ℕ} (N : NumB n) (I : InitB) (Xs : ℝ) : ℝ :=
  N.ρ ^ (2 * N.τ) * max I.iLt (pN N I * N.γb * N.δ0 ^ N.τ) * N.ballSum Xs / (Xs * N.δ0 ^ (2 * N.τ + 1))

/-- `C₃` of route B (iso): the frequency sum, without `γ̄²` (audit iso F3). -/
def c3B {n : ℕ} (N : NumB n) (I : InitB) (Xs : ℝ) : ℝ :=
  N.ρ ^ (2 * N.τ) * max I.iLt (pN N I * N.γb * N.δ0 ^ N.τ) * N.freqSum Xs / (Xs * N.δ0 ^ (2 * N.τ + 1))

/-- The default initial values (`bp:p2s2:defaults`). -/
def defaultInitB (n : ℕ) (σK σB μ : ℝ) : InitB :=
  ⟨2 * n * (σK - μ), (σK - μ) * (σB - μ), 2 * n * (σK - μ) * (σB - μ)⟩

end KAM.Sharp2.RouteB

/-- The frozen G1 body with the refined data of route B and the certificates bound to the frequency (as B9's
`G1BodyA`): the additional hypotheses are `RefHypB` and `N.Cert` on a set `Ω ∋ ω`. -/
def G1BodyB {n : ℕ} (N : NumB n) (I : InitB) (M σK σB σT μ γbar C₁ C₂ : ℝ) : Prop :=
  ∀ (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))) (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω : Fin n → ℝ) (γ e : ℝ) (Ω : Set (Fin n → ℝ)),
    IsOpen 𝔅 →
    (∀ z ∈ 𝔅, ∀ k : Fin n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ 𝔅) →
    DifferentiableOn ℂ h 𝔅 →
    (∀ z ∈ 𝔅, ∀ k : Fin n → ℤ, h (z.1 + (fun i => (k i : ℂ)), z.2) = h z) →
    (∀ θ I : Fin n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ 𝔅 →
      (h ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) →
    (∀ z ∈ 𝔅, ‖fderiv ℂ h z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ h z‖ ≤ M ∧
      ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ M ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ M) →
    DifferentiableOn ℂ K (complexStrip n N.ρ) → IsPrimaryOnStrip K N.ρ → IsRealOnReals K →
    (∀ θ ∈ complexStrip n N.ρ, Metric.ball (K θ) μ ⊆ 𝔅) →
    (∀ θ ∈ complexStrip n N.ρ, ‖fderiv ℂ K θ‖ ≤ σK - μ) →
    (∀ θ ∈ complexStrip n N.ρ, IsUnit (gramMatrix K θ).det ∧
      ∀ i, ∑ j, ‖(gramMatrix K θ)⁻¹ i j‖ ≤ σB - μ) →
    IsUnit (torsionAverage h K ω).det →
    (∀ i, ∑ j, ‖(torsionAverage h K ω)⁻¹ i j‖ ≤ σT - μ) →
    RefHypB N I 𝔅 h K N.ρ → ω ∈ Ω → N.Cert γ Ω →
    0 < γ → γ ≤ γbar → IsDiophantine γ N.τ ω →
    (∀ θ ∈ complexStrip n N.ρ, ‖invarianceError h K ω θ‖ ≤ e) →
    C₁ * e / (γ ^ 4 * N.ρ ^ (4 * N.τ)) < 1 →
    ∃ Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ),
      DifferentiableOn ℂ Kinf (complexStrip n N.ρinf) ∧ IsPrimaryOnStrip Kinf N.ρinf ∧
      IsRealOnReals Kinf ∧
      (∀ θ ∈ complexStrip n N.ρinf, invarianceError h Kinf ω θ = 0) ∧
      (∀ θ ∈ complexStrip n N.ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ 𝔅) ∧
      (∀ θ ∈ complexStrip n N.ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
      ∀ θ ∈ complexStrip n N.ρinf, ‖Kinf θ - K θ‖ ≤ C₂ * e / (γ ^ 2 * N.ρ ^ (2 * N.τ))

/-- Blueprint `bp:p2s2:g1` / `bp:p2s2:frozen-form`, route B, refined: `C₁ = min {C₁♯, c1B}`, `C₂ = max {C₂♯, c2B}`,
by cases on the branch of the minimum (audit F6). Never worse than v1. -/
theorem g1_v2B {n : ℕ} {N : NumB n} (hN : N.Wf) (hfix : N.iso = false) (hτ : (n : ℝ) - 1 ≤ N.τ)
    {M σK σB σT μ γbar Xs : ℝ} (I : InitB) (hM : 0 < M) (hσK : 0 < σK) (hσB : 0 < σB) (hσT : 0 < σT)
    (hμ : 0 < μ) (hγbar : γbar = N.γb) (hsK : N.sK = σK - μ / 2) (hsB : σB - μ < N.sB) (hI : I.Wf N σT μ)
    (hfit : N.Fits I) (hOk : routeBOk N (capB N σT μ 0) (initB I σK σB μ) (μ / 2) Xs) (hXpos : 0 < Xs) :
    0 < min (sharpC1 n N.τ (cohConstR n N.τ) M σK σB σT γbar μ N.τ N.ρ N.ρinf) (c1B N I Xs) ∧
      0 < max (sharpC2 n N.τ (cohConstR n N.τ) M σK σB σT γbar N.τ N.ρ N.ρinf) (c2B N I Xs) ∧
      G1BodyB N I M σK σB σT μ γbar
        (min (sharpC1 n N.τ (cohConstR n N.τ) M σK σB σT γbar μ N.τ N.ρ N.ρinf) (c1B N I Xs))
