/-! ## Part 1. The restart -/

/-- The conclusion of the frozen G1 (`G1Body` after its hypotheses) for `K`, with the closeness
bound `D`. -/
def G1Concl (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))) (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω : Fin n → ℝ) (ρinf μ σK D : ℝ) : Prop :=
  ∃ Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ),
    DifferentiableOn ℂ Kinf (complexStrip n ρinf) ∧ IsPrimaryOnStrip Kinf ρinf ∧
    IsRealOnReals Kinf ∧
    (∀ θ ∈ complexStrip n ρinf, invarianceError h Kinf ω θ = 0) ∧
    (∀ θ ∈ complexStrip n ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ 𝔅) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
    ∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ D

/-- The closeness and the ball of the restart (`step:p2s2hc:rs-ball`). -/
theorem restart_close {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {K K₁ Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ρinf μ D₁ dinf : ℝ}
    (hball : ∀ θ ∈ complexStrip n ρinf, Metric.ball (K θ) μ ⊆ 𝔅)
    (hd : ∀ θ ∈ complexStrip n ρinf, ‖K₁ θ - K θ‖ ≤ dinf) (hbud : dinf + D₁ ≤ μ / 2)
    (h7 : ∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K₁ θ‖ ≤ D₁) :
    (∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ dinf + D₁) ∧
      ∀ θ ∈ complexStrip n ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ 𝔅 := by
  have hc : ∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ dinf + D₁ := fun θ hθ => by
    have e : Kinf θ - K θ = (Kinf θ - K₁ θ) + (K₁ θ - K θ) := by abel
    rw [e, add_comm dinf]
    exact (norm_add_le _ _).trans (add_le_add (h7 θ hθ) (hd θ hθ))
  refine ⟨hc, fun θ hθ z hz => hball θ hθ ?_⟩
  rw [Metric.mem_ball] at hz ⊢
  calc dist z (K θ) ≤ dist z (Kinf θ) + dist (Kinf θ) (K θ) := dist_triangle _ _ _
    _ < μ / 2 + μ / 2 :=
        add_lt_add_of_lt_of_le hz (by rw [dist_eq_norm]; exact (hc θ hθ).trans hbud)
    _ = μ := by ring

/-- Proposition `bp:p2s2hc:restart`: the frozen conclusion for any `K₁` near `K` transfers to
`K`. -/
theorem concl_restart {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {K K₁ : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)}
    {ω : Fin n → ℝ} {ρinf μ μ₁ σK σK₁ D₁ dinf : ℝ}
    (hball : ∀ θ ∈ complexStrip n ρinf, Metric.ball (K θ) μ ⊆ 𝔅)
    (hd : ∀ θ ∈ complexStrip n ρinf, ‖K₁ θ - K θ‖ ≤ dinf) (hbud : dinf + D₁ ≤ μ / 2)
    (hσ : σK₁ - μ₁ / 2 ≤ σK - μ / 2) (h1 : G1Concl 𝔅 h K₁ ω ρinf μ₁ σK₁ D₁) :
    G1Concl 𝔅 h K ω ρinf μ σK (dinf + D₁) := by
  obtain ⟨Kinf, h1, h2, h3, h4, -, h6, h7⟩ := h1
  obtain ⟨hc, hb⟩ := restart_close hball hd hbud h7
  exact ⟨Kinf, h1, h2, h3, h4, hb, fun θ hθ => (h6 θ hθ).trans hσ, hc⟩

/-- The conclusion of the frozen I-G1 (`IG1Body` after its hypotheses) for `(K, s)`, with the
closeness bounds `D` (torus) and `Dω` (frequency). -/
def IG1Concl (D : Set ((Fin n → ℂ) × (Fin n → ℂ))) (H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ωstar : Fin n → ℝ)
    (s E₀ ρinf μ σK σω Dk Dω : ℝ) : Prop :=
  ∃ (Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (sinf : ℝ),
    1 < sinf ∧ sinf < σω ∧ DifferentiableOn ℂ Kinf (complexStrip n ρinf) ∧
    IsPrimaryOnStrip Kinf ρinf ∧ IsRealOnReals Kinf ∧
    (∀ θ ∈ complexStrip n ρinf, invarianceError H Kinf (sinf • ωstar) θ = 0) ∧
    (∀ θ ∈ complexStrip n ρinf, H (Kinf θ) = (E₀ : ℂ)) ∧
    (∀ θ ∈ complexStrip n ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ D) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ Dk) ∧
    ‖sinf • ωstar - s • ωstar‖ ≤ Dω

/-- Proposition `bpI:p2s2hc:restart`. -/
theorem iconcl_restart {D : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {K K₁ : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)}
    {ωstar : Fin n → ℝ} {s s₁ E₀ ρinf μ μ₁ σK σK₁ σω D₁ D₁ω dinf dω : ℝ}
    (hball : ∀ θ ∈ complexStrip n ρinf, Metric.ball (K θ) μ ⊆ D)
    (hd : ∀ θ ∈ complexStrip n ρinf, ‖K₁ θ - K θ‖ ≤ dinf) (hdω : ‖s₁ • ωstar - s • ωstar‖ ≤ dω)
    (hbud : dinf + D₁ ≤ μ / 2) (hσ : σK₁ - μ₁ / 2 ≤ σK - μ / 2)
    (h1 : IG1Concl D H K₁ ωstar s₁ E₀ ρinf μ₁ σK₁ σω D₁ D₁ω) :
    IG1Concl D H K ωstar s E₀ ρinf μ σK σω (dinf + D₁) (dω + D₁ω) := by
  obtain ⟨Kinf, sinf, hs1, hs2, h1, h2, h3, h4, h5, -, h7, h8, h9⟩ := h1
  obtain ⟨hc, hb⟩ := restart_close hball hd hbud h8
  refine ⟨Kinf, sinf, hs1, hs2, h1, h2, h3, h4, h5, hb, fun θ hθ => (h7 θ hθ).trans hσ, hc, ?_⟩
  have e : sinf • ωstar - s • ωstar = (sinf • ωstar - s₁ • ωstar) + (s₁ • ωstar - s • ωstar) := by
    abel
  rw [e, add_comm dω]
  exact (norm_add_le _ _).trans (add_le_add h9 hdω)

/-! ## Part 2. The transfer lemmas -/

/-- Definition `bp:p2s2hc:lin`: `Ẽ = E + DX_h(K) ΔK - DΔK ω`. At `ω = s₁ ω*` it is the
isoenergetic `Ẽ_s` of `bpI:p2s2hc:lin`. -/
def linNewError (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω : Fin n → ℝ) (θ : Fin n → ℂ) :
    (Fin n → ℂ) × (Fin n → ℂ) :=
  invarianceError h K ω θ + fderiv ℂ (hamiltonianVectorFieldℂ h) (K θ) (ΔK θ) -
    fderiv ℂ ΔK θ (fun i => (ω i : ℂ))

/-- `bpI:p2s2hc:lin`: the linearised energy error `⟨H∘K⟩ - E₀ + ⟨DH(K) ΔK⟩`. -/
def linNewEnergy (H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (E₀ : ℝ) : ℂ :=
  KAM.Iso.energyAverage H K - (E₀ : ℂ) +
    ∫ θ in Set.pi Set.univ (fun _ : Fin n => Set.Icc (0 : ℝ) 1),
      fderiv ℂ H (K (fun l => (θ l : ℂ))) (ΔK (fun l => (θ l : ℂ)))

/-- A step (Assumption `bp:p2s2hc:standing`(d)): holomorphic on `T_{ρ₀}`, `1`-periodic, real. -/
structure IsStep (ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ρ₀ : ℝ) : Prop where
  holo : DifferentiableOn ℂ ΔK (complexStrip n ρ₀)
  periodic : ∀ θ ∈ complexStrip n ρ₀, ∀ k : Fin n → ℤ, ΔK (θ + fun i => (k i : ℂ)) = ΔK θ
  real : IsRealOnReals ΔK

/-- Lemma `bp:p2s2hc:struct`. -/
theorem step_struct {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ρ₀ ρ₁ μ μ₁ d₀ d₁ ℓ : ℝ}
    (hK : DifferentiableOn ℂ K (complexStrip n ρ₀)) (hKp : IsPrimaryOnStrip K ρ₀)
    (hKr : IsRealOnReals K) (hball : ∀ θ ∈ complexStrip n ρ₀, Metric.ball (K θ) μ ⊆ 𝔅)
    (hDK : ∀ θ ∈ complexStrip n ρ₀, ‖fderiv ℂ K θ‖ ≤ ℓ) (hS : IsStep ΔK ρ₀) (hρ : ρ₁ ≤ ρ₀)
    (hd0 : ∀ θ ∈ complexStrip n ρ₁, ‖ΔK θ‖ ≤ d₀)
    (hd1 : ∀ θ ∈ complexStrip n ρ₁, ‖fderiv ℂ ΔK θ‖ ≤ d₁) (hμ : d₀ + μ₁ ≤ μ) :
    DifferentiableOn ℂ (K + ΔK) (complexStrip n ρ₁) ∧ IsPrimaryOnStrip (K + ΔK) ρ₁ ∧
      IsRealOnReals (K + ΔK) ∧ (∀ θ ∈ complexStrip n ρ₁, Metric.ball ((K + ΔK) θ) μ₁ ⊆ 𝔅) ∧
      ∀ θ ∈ complexStrip n ρ₁, ‖fderiv ℂ (K + ΔK) θ‖ ≤ ℓ + d₁ := by
  sorry

/-- Lemma `bp:p2s2hc:error`: `|E₁| ≤ ℓ_E + (M₂/2) d₀²` (Taylor, `KAM.taylor_iteratedFDeriv`). -/
theorem invarianceError_step_le {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))}

/-- The numbers of Lemma `bp:p2s2hc:frames`: initial values of `K` and the increments of `DΔK`. -/
structure FrameNum where
  ℓ : ℝ
  ℓt : ℝ
  β : ℝ
  ν : ℝ
  νt : ℝ
  d1 : ℝ
  d1t : ℝ

namespace FrameNum

variable (F : FrameNum)

/-- `g = ℓᵗ d₁ + d₁ᵗ ℓ + d₁ᵗ d₁ ≥ |G₁ - G|`. -/
def g : ℝ := F.ℓt * F.d1 + F.d1t * F.ℓ + F.d1t * F.d1
/-- `β₁ = β/(1 - βg) ≥ |B₁|`. -/
def β1 : ℝ := F.β / (1 - F.β * F.g)
/-- `b = β² g/(1 - βg) ≥ |B₁ - B|`. -/
def b : ℝ := F.β ^ 2 * F.g / (1 - F.β * F.g)
/-- `n_Δ = β₁ d₁ + ℓ b ≥ |N₁ - N|`. -/
def nΔ : ℝ := F.β1 * F.d1 + F.ℓ * F.b
/-- `n_Δᵗ = β₁ d₁ᵗ + ℓᵗ b ≥ |N₁ᵀ - Nᵀ|`. -/
def nΔt : ℝ := F.β1 * F.d1t + F.ℓt * F.b

end FrameNum

/-- Lemma `bp:p2s2hc:frames` (pointwise). -/
theorem frames_step {K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {θ : Fin n → ℂ}
    {F : FrameNum} (hK : DifferentiableAt ℂ K θ) (hΔ : DifferentiableAt ℂ ΔK θ)
    (hG : IsUnit (gramMatrix K θ).det) (hL : ‖tangentMat K θ‖ ≤ F.ℓ)
    (hLt : ‖(tangentMat K θ)ᵀ‖ ≤ F.ℓt) (hB : ‖(gramMatrix K θ)⁻¹‖ ≤ F.β)
    (hN : ‖normalMat K θ‖ ≤ F.ν) (hNt : ‖(normalMat K θ)ᵀ‖ ≤ F.νt)
    (h1 : ‖tangentMat ΔK θ‖ ≤ F.d1) (h1t : ‖(tangentMat ΔK θ)ᵀ‖ ≤ F.d1t)
    (hs : F.β * F.g < 1) :
    IsUnit (gramMatrix (K + ΔK) θ).det ∧ ‖(tangentMat (K + ΔK) θ)ᵀ‖ ≤ F.ℓt + F.d1t ∧
      ‖(gramMatrix (K + ΔK) θ)⁻¹‖ ≤ F.β1 ∧
      ‖(gramMatrix (K + ΔK) θ)⁻¹ - (gramMatrix K θ)⁻¹‖ ≤ F.b ∧
      ‖normalMat (K + ΔK) θ - normalMat K θ‖ ≤ F.nΔ ∧
      ‖(normalMat (K + ΔK) θ)ᵀ - (normalMat K θ)ᵀ‖ ≤ F.nΔt ∧
      ‖normalMat (K + ΔK) θ‖ ≤ F.ν + F.nΔ ∧ ‖(normalMat (K + ΔK) θ)ᵀ‖ ≤ F.νt + F.nΔt := by
  sorry

/-- `Δ_ab` of Lemma `bp:p2s2hc:torsion`(a). -/
def dAB (F : FrameNum) (cTh cDTh d : ℝ) : ℝ :=
  F.nΔt * cTh * (F.ν + F.nΔ) + F.νt * (cDTh * d) * (F.ν + F.nΔ) + F.νt * cTh * F.nΔ

/-- Lemma `bp:p2s2hc:torsion`(a), the frame part (pointwise; `KAM.Sharp2.torsionAB`). -/
theorem torsionAB_step_sub_le {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ}
    {K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {θ : Fin n → ℂ} {F : FrameNum}
    {cTh cDTh d : ℝ} (hG : IsUnit (gramMatrix K θ).det)
    (hG₁ : IsUnit (gramMatrix (K + ΔK) θ).det) (hN : ‖normalMat K θ‖ ≤ F.ν)
    (hNt : ‖(normalMat K θ)ᵀ‖ ≤ F.νt) (hdN : ‖normalMat (K + ΔK) θ - normalMat K θ‖ ≤ F.nΔ)
    (hdNt : ‖(normalMat (K + ΔK) θ)ᵀ - (normalMat K θ)ᵀ‖ ≤ F.nΔt)
    (hc : ‖hamComm h (K θ)‖ ≤ cTh) (hc₁ : ‖hamComm h ((K + ΔK) θ)‖ ≤ cTh)
    (hcd : ‖hamComm h ((K + ΔK) θ) - hamComm h (K θ)‖ ≤ cDTh * d) :
    ‖torsionAB h (K + ΔK) θ - torsionAB h K θ‖ ≤ dAB F cTh cDTh d := by
  sorry

/-- Lemma `bp:p2s2hc:torsion`(c) (pointwise): the two sup forms and the increment. The local parts
`l₀`, `l₁` come from `KAM.Sharp2.norm_torsion_sub_torsionAB_le` at `K` (frequency `ω`) and at `K₁`
(frequency `ω₁`; `ω₁ = ω` at fixed frequency, `s₁ ω*` against `s ω*` in the iso form: the frame
part `𝒯_ab` does not depend on the frequency). -/
theorem torsion_step_le {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ}
    {K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ω ω₁ : Fin n → ℝ} {θ : Fin n → ℂ}
    {t dab l₀ l₁ a₁ : ℝ} (hT : ‖torsion h K ω θ‖ ≤ t)
    (hab : ‖torsionAB h (K + ΔK) θ - torsionAB h K θ‖ ≤ dab)
    (hab₁ : ‖torsionAB h (K + ΔK) θ‖ ≤ a₁)
    (hl₀ : ‖torsion h K ω θ - torsionAB h K θ‖ ≤ l₀)
    (hl₁ : ‖torsion h (K + ΔK) ω₁ θ - torsionAB h (K + ΔK) θ‖ ≤ l₁) :
    ‖torsion h (K + ΔK) ω₁ θ‖ ≤ min (t + dab + l₀) a₁ + l₁ ∧
      ‖torsion h (K + ΔK) ω₁ θ - torsion h K ω θ‖ ≤ dab + l₀ + l₁ := by
  sorry

/-- Lemma `bpI:p2s2hc:errors`(b): the new energy error (Taylor for `H` at real points). -/
theorem energyAverage_step_le {D : Set ((Fin n → ℂ) × (Fin n → ℂ))}
    {H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)}
    {E₀ μ Mh2 d₀R ℓH : ℝ} (hopen : IsOpen D) (hH : DifferentiableOn ℂ H D)
    (hd2 : ∀ z ∈ D, ‖iteratedFDeriv ℂ 2 H z‖ ≤ Mh2)
    (hball : ∀ θ : Fin n → ℝ, Metric.ball (K fun l => (θ l : ℂ)) μ ⊆ D)
    (hd0 : ∀ θ : Fin n → ℝ, ‖ΔK fun l => (θ l : ℂ)‖ ≤ d₀R) (hd0μ : d₀R < μ)
    (hint : IntegrableOn (fun θ : Fin n → ℝ => H (K fun l => (θ l : ℂ)))

structure StepBounds (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (K ΔK : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω ω₁ : Fin n → ℝ) (ρ₁ ρinf ρT : ℝ)
    (b : StepNum) : Prop where
  d0 : ∀ θ ∈ complexStrip n ρ₁, ‖ΔK θ‖ ≤ b.d0
  dinf : ∀ θ ∈ complexStrip n ρinf, ‖ΔK θ‖ ≤ b.dinf
  d1 : ∀ θ ∈ complexStrip n ρ₁, ‖tangentMat ΔK θ‖ ≤ b.d1 ∧ ‖(tangentMat ΔK θ)ᵀ‖ ≤ b.d1t
  real : ∀ θ : Fin n → ℝ, ‖ΔK fun l => (θ l : ℂ)‖ ≤ b.d0R ∧
    ‖tangentMat ΔK (fun l => (θ l : ℂ))‖ ≤ b.d1R ∧ ‖(tangentMat ΔK (fun l => (θ l : ℂ)))ᵀ‖ ≤ b.d1tR
  lin : ∀ θ ∈ complexStrip n ρ₁, ‖linNewError h K ΔK ω₁ θ‖ ≤ b.lE
  loc0 : ∀ θ ∈ complexStrip n ρT, ‖errDerivMat h K ω θ‖ ≤ b.dE0 ∧ ‖lagrangeMat K θ‖ ≤ b.o0
  loc0R : ∀ θ : Fin n → ℝ, ‖errDerivMat h K ω (fun l => (θ l : ℂ))‖ ≤ b.dE0R ∧
    ‖lagrangeMat K (fun l => (θ l : ℂ))‖ ≤ b.o0R
  loc1 : ∀ θ ∈ complexStrip n ρT, ‖lagrangeMat (K + ΔK) θ‖ ≤ b.o1
  loc1R : ∀ θ : Fin n → ℝ, ‖lagrangeMat (K + ΔK) (fun l => (θ l : ℂ))‖ ≤ b.o1R

namespace StepNum

variable (b : StepNum) (k : KData)

/-- `e₁ = ℓ_E + (M₂/2) d₀²` (Lemma `bp:p2s2hc:error`). -/
def e1 (M2 : ℝ) : ℝ := b.lE + M2 / 2 * b.d0 ^ 2
/-- The frame numbers on the strip. -/
def frS : FrameNum := ⟨k.ℓ, k.ℓt, k.β, k.ν, k.νt, b.d1, b.d1t⟩
/-- The frame numbers on the real torus. -/
def frR : FrameNum := ⟨k.ℓ, k.ℓt, k.β, k.ν, k.νt, b.d1R, b.d1tR⟩
/-- A bound of the matrix `W` of the local torsion part (`bp:p2s2hc:torsion`). -/
def wB (nn M1 ℓ ℓt dE : ℝ) : ℝ := ℓt * (4 * nn * M1) * ℓ + dE * (2 * nn * ℓ + ℓt)
/-- `λ₀`, the local part of `K`'s torsion, on the strip `T'` (`real = false`) or on `ℝⁿ`. -/
def lam0 (nn M1 : ℝ) (real : Bool) : ℝ :=
  let dE := if real then b.dE0R else b.dE0
  let o := if real then b.o0R else b.o0
  k.νt * k.β * dE + k.β ^ 3 * wB nn M1 k.ℓ k.ℓt dE * o
/-- `λ₁`, the local part of `K₁`'s torsion, with `ε₁' = n e₁/δ` (Cauchy). -/
def lam1 (nn M1 M2 δ : ℝ) (real : Bool) : ℝ :=
  let F := if real then b.frR k else b.frS k
  let dE := nn * δ⁻¹ * b.e1 M2
  let o := if real then b.o1R else b.o1
  (F.νt + F.nΔt) * F.β1 * dE + F.β1 ^ 3 * wB nn M1 (F.ℓ + F.d1) (F.ℓt + F.d1t) dE * o
/-- `t₁`, the torsion bound of `K₁` on `T'` (Lemma `bp:p2s2hc:torsion`(c)). -/
def t1 (nn M1 M2 δ cTh : ℝ) : ℝ :=
  min (k.t + dAB (b.frS k) cTh (2 * M2) b.d0 + b.lam0 k nn M1 false)
      (((b.frS k).νt + (b.frS k).nΔt) * cTh * ((b.frS k).ν + (b.frS k).nΔ)) +
    b.lam1 k nn M1 M2 δ false
/-- `t_A ≥ |⟨T₁⟩ - ⟨T⟩|` (Lemma `bp:p2s2hc:torsion`(d)). -/
def tA (nn M1 M2 δ cTh : ℝ) : ℝ :=
  dAB (b.frR k) cTh (2 * M2) b.d0R + b.lam0 k nn M1 true + b.lam1 k nn M1 M2 δ true
/-- The transferred initial values of `K₁` (Definition `bp:p2s2hc:data`). -/
def init1 (nn M1 M2 δ cTh : ℝ) : InitA :=
  ⟨k.ℓt + b.d1t, k.ν + (b.frS k).nΔ, k.νt + (b.frS k).nΔt, b.t1 k nn M1 M2 δ cTh⟩
/-- `σ_{K,1} = ℓ + d₁ + μ₁`. -/
def sK1 (μ₁ : ℝ) : ℝ := k.ℓ + b.d1 + μ₁
/-- `σ_{B,1} = β₁ + μ₁`. -/
def sB1 (μ₁ : ℝ) : ℝ := (b.frS k).β1 + μ₁
/-- `σ_{T,1} = m/(1 - m t_A) + μ₁`. -/
def sT1 (nn M1 M2 δ cTh μ₁ : ℝ) : ℝ := k.m / (1 - k.m * b.tA k nn M1 M2 δ cTh) + μ₁

end StepNum

/-- The hybrid condition `HybOk_A'` of Definition `bp:p2s2hc:data` for `K` with the data `k`, the
step numbers `b`, the restart `N₁ = N.restart R`, margin `μ₁`, `X⋆`, budgets `bF`, `bT`. -/
structure HybOkR (N : NumA) (R : RestartA) (k : KData) (b : StepNum)
    (σK σB σT μ μ₁ γ Xs bF bT cTh : ℝ) : Prop where
  wf : (N.restart R).Wf
  μ₁_pos : 0 < μ₁
  ρ_le : R.ρ₁ ≤ N.ρ
  ℓ : k.ℓ = σK - μ
  β : k.β = σB - μ
  m : k.m = σT - μ

  budget : Ref.Budget (N.restart R) (b.sT1 k N.n N.M1 N.M2 ((N.restart R).δ 0) cTh μ₁ - μ₁)
      (b.init1 k N.n N.M1 N.M2 ((N.restart R).δ 0) cTh).iH bF bT
  Xs_pos : 0 < Xs
  m1 : b.d0 + μ₁ ≤ μ
  m2 : b.d1 + μ₁ / 2 ≤ μ / 2
  gS : k.β * (b.frS k).g < 1
  gR : k.β * (b.frR k).g < 1
  tA : k.m * b.tA k N.n N.M1 N.M2 ((N.restart R).δ 0) cTh < 1
  small : b.e1 N.M2 / (γ ^ 4 * delta0 3 R.ρ₁ N.ρinf ^ (2 * (2 * N.τ))) < Xs
  bud : b.dinf + c2A (N.restart R) Xs * b.e1 N.M2 / (γ ^ 2 * R.ρ₁ ^ (2 * N.τ)) ≤ μ / 2

/-- Corollary `bp:p2s2hc:frozen`, route A with L8′: the plain branch (`g1_v2A'`'s constants at

theorem g1_v2A'_hybrid (N : NumA) (hN : N.Wf) (hfix : N.iso = false) (hΘ : N.Θ = 0)
    (hτ : (N.n : ℝ) - 1 ≤ N.τ) {M σK σB σT μ γbar Xs bF bT : ℝ} (I : InitA) (hM : 0 < M)
    (hσK : 0 < σK) (hσB : 0 < σB) (hσT : 0 < σT) (hμ : 0 < μ) (hγbar : γbar = N.γb)
    (hsK : N.sK = σK - μ / 2) (hOk : routeAOkR N bF (N.capR (μ / 2) bT) (initR I σK σB μ) Xs)
    (hB : Ref.Budget N (σT - μ) I.iH bF bT) (hXpos : 0 < Xs) :
    ∀ (𝔅 : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ))) (h : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ)
      (K ΔK : (Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ)) (ω : Fin N.n → ℝ) (γ e : ℝ)
      (Ω : Set (Fin N.n → ℝ)) (R : RestartA) (k : KData) (b : StepNum)
      (μ₁ X₁ bF₁ bT₁ cTh : ℝ),
      IsOpen 𝔅 →
      (∀ z ∈ 𝔅, ∀ k : Fin N.n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ 𝔅) →
      DifferentiableOn ℂ h 𝔅 →
      (∀ z ∈ 𝔅, ∀ k : Fin N.n → ℤ, h (z.1 + (fun i => (k i : ℂ)), z.2) = h z) →
      (∀ θ I : Fin N.n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ 𝔅 →
        (h ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) →
      (∀ z ∈ 𝔅, ‖fderiv ℂ h z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ h z‖ ≤ M ∧
        ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ M ∧
        ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ M) →
      DifferentiableOn ℂ K (complexStrip N.n N.ρ) → IsPrimaryOnStrip K N.ρ → IsRealOnReals K →
      (∀ θ ∈ complexStrip N.n N.ρ, Metric.ball (K θ) μ ⊆ 𝔅) →
      (∀ θ ∈ complexStrip N.n N.ρ, ‖fderiv ℂ K θ‖ ≤ σK - μ) →
      (∀ θ ∈ complexStrip N.n N.ρ, IsUnit (gramMatrix K θ).det ∧
        ∀ i, ∑ j, ‖(gramMatrix K θ)⁻¹ i j‖ ≤ σB - μ) →
      IsUnit (torsionAverage h K ω).det →
      (∀ i, ∑ j, ‖(torsionAverage h K ω)⁻¹ i j‖ ≤ σT - μ) →
      RefHypA N I 𝔅 h K ω → ω ∈ Ω → N.Cert γ Ω →
      0 < γ → γ ≤ γbar → IsDiophantine γ N.τ ω →
      (∀ θ ∈ complexStrip N.n N.ρ, ‖invarianceError h K ω θ‖ ≤ e) →
      -- the hybrid data (used in the second branch only)
      (∀ z ∈ 𝔅, ‖hamComm h z‖ ≤ cTh) → I.iLt ≤ k.ℓt → I.iN ≤ k.ν → I.iNt ≤ k.νt →
      I.iH ≤ k.t → IsStep ΔK N.ρ →
      StepBounds h K ΔK ω ω R.ρ₁ N.ρinf (R.ρ₁ - (N.restart R).δ 0) b →
      (N.restart R).Cert γ Ω →
      (min (sharpC1 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar μ N.τ N.ρ N.ρinf) (c1A N Xs) *
          e / (γ ^ 4 * N.ρ ^ (4 * N.τ)) < 1 ∨
        HybOkR N R k b σK σB σT μ μ₁ γ X₁ bF₁ bT₁ cTh) →
      G1Concl 𝔅 h K ω N.ρinf μ σK
        (max (max (sharpC2 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar N.τ N.ρ N.ρinf)
            (c2A N Xs) * e / (γ ^ 2 * N.ρ ^ (2 * N.τ)))
          (b.dinf + c2A (N.restart R) X₁ * b.e1 N.M2 / (γ ^ 2 * R.ρ₁ ^ (2 * N.τ)))) := by
  sorry
