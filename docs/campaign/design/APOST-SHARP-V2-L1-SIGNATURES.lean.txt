structure SDConst (n : ℕ) (κ γ δ c : ℝ) (Ω : Set (Fin n → ℝ)) : Prop where
  c_pos : 0 < c
  delta_pos : 0 < δ
  sol : ∀ ω ∈ Ω, ∀ {r : ℝ}, δ < r → ∀ {p : ℕ} (g : (Fin n → ℂ) → Fin p → ℂ) (H : ℝ),
    DifferentiableOn ℂ g (complexStrip n r) → IsPeriodicOn g r →
    (∀ θ ∈ complexStrip n r, ‖g θ‖ ≤ H) → avg g = 0 →
    ∀ θ ∈ complexStrip n (r - δ), ‖cohR ω g θ‖ ≤ c * γ⁻¹ * δ ^ (-κ) * H
  apriori : ∀ ω ∈ Ω, ∀ {r : ℝ}, δ < r → ∀ {p q : ℕ}
    (u : (Fin n → ℂ) → Matrix (Fin p) (Fin q) ℂ) (H : ℝ),
    DifferentiableOn ℂ u (complexStrip n r) → IsPeriodicOn u r → avg u = 0 →
    (∀ θ ∈ complexStrip n r, ‖fderiv ℂ u θ (fun l => (ω l : ℂ))‖ ≤ H) →
    ∀ θ ∈ complexStrip n (r - δ), ‖u θ‖ ≤ c * γ⁻¹ * δ ^ (-κ) * H
  real : ∀ ω ∈ Ω, ∀ {r : ℝ}, δ ≤ r → ∀ {p : ℕ} (g : (Fin n → ℂ) → Fin p → ℂ) (H : ℝ),
    DifferentiableOn ℂ g (complexStrip n r) → IsPeriodicOn g r →
    (∀ θ ∈ complexStrip n r, ‖g θ‖ ≤ H) → avg g = 0 →
    ∀ θ : Fin n → ℝ, ‖cohR ω g (fun i => (θ i : ℂ))‖ ≤ c * γ⁻¹ * δ ^ (-κ) * H

def ray {n : ℕ} (ωstar : Fin n → ℝ) : Set (Fin n → ℝ) := {ω | ∃ s : ℝ, 1 ≤ s ∧ ω = s • ωstar}

def stepConsts (p : StepIn) : StepConsts :=
  let n : ℝ := p.n
  let g := p.γb * p.δ ^ p.κ
  let g1 := g * p.δ
  let sKt := (1 + p.θ) * p.sK
  let CΩ := 2 * (n - 1) * p.c1
  let CEL := (p.sK + p.sN * p.q) / (1 - p.q ^ 2)
  let CEN := (p.sN + p.sK * p.q) / (1 - p.q ^ 2)
  let CE := p.lam * CEL + g * CEN
  let CDE := n * (p.sK + p.sN) * (max p.lam g + (p.sLt + p.sNt) * CE) / (1 - p.q)
  let CNN := n * (p.lam + p.sNt * CE)
  let avgB := p.cbar * (p.δ / p.r) ^ p.κ
  let C0 := if p.iso then p.sT * max (p.lam + p.t * avgB) (p.lamW + p.y0 * avgB) else p.sT * (p.lam + p.t * avgB)
  let CxiN := C0 + p.c
  let CKt := p.sN * CxiN
  let CDLt := n * p.sN * (C0 + p.c1)
  let CDLtT := 2 * n * p.sN * (C0 + p.c1)
  let CdN := (p.M1 * p.sK + n) * p.sB + p.sN * ((p.sLt * p.M1 + 2 * n) * p.sK + p.sLt * (p.M1 * p.sK + n)) * p.sB
  let CXN := p.M1 * p.sN + CdN
  let CEt := CE + CXN * CxiN + p.sN * g + p.M2 / 2 * CKt + (if p.iso then sKt * p.Θ * C0 else 0)
  let QtN := CDLtT * CEt + CNN * CxiN + p.δ * p.sLt * (p.M2 / 2) * CKt ^ 2
  let CgL := p.lam + p.t * CxiN + (if p.iso then p.Θ * C0 else 0)
  let CxiL := p.c * CgL
  let CDK := sKt * CxiL + g * CKt
  let CEb := CEt + sKt * CgL
  let CG := n * p.sLt + 2 * n * p.sK
  let CBb := p.sB ^ 2 * CG * CDK / (1 - p.θ)
  let CNb := n * CDK * p.sB + p.sK * CBb
  let CNbT := CBb * p.sLt + 2 * n * CDK * p.sB
  let w := 2 * p.M1 * p.sLt * p.sK + 2 * n * p.sK + n * p.sLt
  let CDT0 := 2 * p.M1 * (CNbT * p.sN + p.sNt * CNb) + 2 * p.M2 * p.sNt * p.sN * CDK * p.δ +
    g * (2 * p.sNt * p.sB * CDE + 2 * p.sB ^ 3 * w * CΩ)
  let CDT := if p.iso then max (CDT0 + p.Θ * C0 * g1) (n * p.Θ * C0 * g1 + 2 * n * g1) else CDT0
  let QbN := (1 + n * p.θξ) * (1 + p.θΩ) * QtN +
    (if p.iso then (1 + n * p.θξ) * p.Θ * CΩ * QtN * C0 * p.θξ / CxiL else 0)
  let QL0 := p.sNt * (p.M2 / 2) * CKt ^ 2 * g1 + p.sB ^ 2 * CΩ * g ^ 2 + n * p.sNt * CEt * CxiL +
    p.sNt * CgL * (n * sKt * CxiL + g * CDLt) + CNbT * CEb
  let QL := if p.iso then max (QL0 + C0 * CgL * g1 + C0 * p.sNt * CDLt * p.Θ * g)
      (p.Mh2 / 2 * CKt ^ 2 * g1 + n * QtN * p.θξ * (p.Θ * CΩ + g1)) else QL0
  ⟨CΩ, CE, CDE, CNN, C0, CxiN, CKt, CDLt, CDLtT, CEt, QtN, CgL, CxiL, CDK, CEb, CG, CBb, CNb, CNbT, CDT, QbN, QL,
    max QL (2 ^ p.κ * QbN)⟩

def stepCond (p : StepIn) (X : ℝ) : Prop :=
  let k := stepConsts p
  let g := p.γb * p.δ ^ p.κ
  max 1 (p.sB ^ 2) * k.CΩ * g * X ≤ p.q ∧ k.CDE * g * X ≤ p.n ∧ k.CxiL * X ≤ p.θξ ∧
    k.CΩ * k.CgL * X ≤ p.θΩ ∧ k.CKt * (g * p.δ) * X ≤ 1 ∧ k.CDLt * g * X ≤ p.θ * p.sK ∧
    k.CDLtT * g * X ≤ p.θ * p.sLt ∧ p.sB * k.CG * k.CDK * X ≤ p.θ ∧ p.sT * k.CDT * X ≤ p.θ ∧
    2 ^ (p.κ + 1) * k.Q * X ≤ 1

variable {n : ℕ}

def modXiN0 (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) : Fin n → ℂ :=
  (torsionAverage h K ω)⁻¹ *ᵥ avg (fun θ => etaL h K ω θ + torsion h K ω θ *ᵥ cohR ω (etaN h K ω) θ)

def modXiN (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : Fin n → ℂ :=
  modXiN0 h K ω - cohR ω (etaN h K ω) θ

def modKtilde (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : (Fin n → ℂ) × (Fin n → ℂ) :=
  K θ + (sumVec n).symm (normalMat K θ *ᵥ modXiN h K ω θ)

def modXiL (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (θ : Fin n → ℂ) : Fin n → ℂ :=
  -cohR ω (fun θ' => etaL h K ω θ' - torsion h K ω θ' *ᵥ modXiN h K ω θ') θ

open Classical in
def modNewton (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ))
    (ω : Fin n → ℝ) (w : ℝ) : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ) :=
  fun θ => if θ ∈ complexStrip n w then modKtilde h K ω (θ + modXiL h K ω θ) else K θ

end ModStep

def xSeq (κ : ℝ) (Q : ℕ → ℝ) (X0 : ℝ) : ℕ → ℝ
  | 0 => X0
  | s + 1 => 2 ^ (κ + 1) * Q s * xSeq κ Q X0 s ^ 2

structure StageScheme (n k : ℕ) (α : Type*) where
  tor : α → (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)
  par : α → ℝ
  Adm : ℝ → ℝ → ℝ → (Fin k → ℝ) → α → Prop
  step : ℝ → ℝ → α → α

structure StageConsts (k : ℕ) where
  Q : ℝ
  cond : ℝ → Prop
  CDK : ℝ
  Cp : ℝ
  rate : Fin k → ℝ
  wt : Fin k → ℝ

structure StageScheme.IsNewton {n k : ℕ} {α : Type*} (S : StageScheme n k α) (κ γ δmax : ℝ)
    (C : ℝ → StageConsts k) (σ : Fin k → ℝ) : Prop where
  regular : ∀ {r β μ m x}, S.Adm r β μ m x → 0 ≤ μ ∧
    DifferentiableOn ℂ (S.tor x) (complexStrip n r) ∧
    IsPrimaryOnStrip (S.tor x) r ∧ IsRealOnReals (S.tor x)
  step : ∀ {r β μ δ m x}, S.Adm r β μ m x → 0 < δ → δ ≤ δmax → 3 * δ < r →
    (C δ).cond (μ / (γ * δ ^ (κ + 1))) → (C δ).CDK * μ / (γ * δ ^ κ) < β →
    (∀ i, m i + (C δ).rate i * (C δ).wt i * (μ / (γ * δ ^ (κ + 1))) ≤ σ i) →
    S.Adm (r - 3 * δ) (β - (C δ).CDK * μ / (γ * δ ^ κ)) ((C δ).Q * μ ^ 2 / (γ * δ ^ (κ + 1)))
        (fun i => m i + (C δ).rate i * (C δ).wt i * (μ / (γ * δ ^ (κ + 1)))) (S.step r δ x) ∧
      (∀ θ ∈ complexStrip n (r - 3 * δ), ‖S.tor (S.step r δ x) θ - S.tor x θ‖ ≤
        (C δ).CDK * μ / (γ * δ ^ κ)) ∧
      |S.par (S.step r δ x) - S.par x| ≤ (C δ).Cp * μ

abbrev bite0 (ρ ρinf : ℝ) : ℝ := min 1 ((ρ - ρinf) / 6)
abbrev biteS (ρ ρinf : ℝ) (s : ℕ) : ℝ := bite0 ρ ρinf / 2 ^ s
abbrev stripS (ρ ρinf : ℝ) (j : ℕ) : ℝ := ρ - 6 * bite0 ρ ρinf * (1 - (2 : ℝ)⁻¹ ^ j)

def stageBall {k : ℕ} (κ γ ρ ρinf μ₀ : ℝ) (C : ℝ → StageConsts k) (kpre : ℕ)
    (Ctail : StageConsts k) : ℝ :=
  (∑ s ∈ Finset.range kpre, (C (min 1 ((ρ - ρinf) / 6) / 2 ^ s)).CDK *
      (min 1 ((ρ - ρinf) / 6) / 2 ^ s) *
      xSeq κ (fun s => (C (min 1 ((ρ - ρinf) / 6) / 2 ^ s)).Q)
        (μ₀ / (γ * min 1 ((ρ - ρinf) / 6) ^ (κ + 1))) s) +
    2 * Ctail.CDK * (min 1 ((ρ - ρinf) / 6) / 2 ^ kpre) *
      xSeq κ (fun s => (C (min 1 ((ρ - ρinf) / 6) / 2 ^ s)).Q)
        (μ₀ / (γ * min 1 ((ρ - ρinf) / 6) ^ (κ + 1))) kpre

      IsRealOnReals Kinf ∧

        S.step (ρ - 6 * min 1 ((ρ - ρinf) / 6) *
        (1 - (2 : ℝ)⁻¹ ^ j)) (min 1 ((ρ - ρinf) / 6) / 2 ^ j) x) s) ∧ ρinf < r ∧

        S.step (ρ - 6 * min 1 ((ρ - ρinf) / 6) *
        (1 - (2 : ℝ)⁻¹ ^ j)) (min 1 ((ρ - ρinf) / 6) / 2 ^ j) x) s)) Kinf atTop

        stageBall κ γ ρ ρinf μ₀ C kpre Ctail) := sorry

/-- The Fourier coefficients on the real torus of the component `i` of `f`. -/
abbrev coeffR {ι : Type*} (f : (Fin n → ℂ) → ι → ℂ) (i : ι) : (Fin n → ℤ) → ℂ :=
  torusCoeff fun x => f (fun j => (x j : ℂ)) i

/-- Blueprint `bp:p2s2l1:norms` (vectors): `‖f‖_{1,r} ≤ A`, componentwise, with summability and `0 ≤ A`
(the vector premise of `KAM.cohomological_l1`). -/
def L1Vec {ι : Type*} (r : ℝ) (f : (Fin n → ℂ) → ι → ℂ) (A : ℝ) : Prop :=
  0 ≤ A ∧ ∀ i, Fourier.SummableWeighted (2 * Real.pi * r) (coeffR f i) ∧
    Fourier.weightedNorm (2 * Real.pi * r) (coeffR f i) ≤ A

/-- Blueprint `bp:p2s2l1:norms` (matrices): every row sum of the entry norms is `≤ A`. -/
def L1Mat {ι κ : Type*} [Fintype κ] (r : ℝ) (M : (Fin n → ℂ) → Matrix ι κ ℂ) (A : ℝ) : Prop :=
  0 ≤ A ∧ (∀ i j, Fourier.SummableWeighted (2 * Real.pi * r) (torusCoeff fun x => M (fun l => (x l : ℂ)) i j)) ∧
    ∀ i, ∑ j, Fourier.weightedNorm (2 * Real.pi * r) (torusCoeff fun x => M (fun l => (x l : ℂ)) i j) ≤ A

/-- The Jacobian matrix `(∂_j f_i)`. -/
def jac {ι : Type*} (f : (Fin n → ℂ) → ι → ℂ) (θ : Fin n → ℂ) : Matrix ι (Fin n) ℂ :=
  Matrix.of fun i j => fderiv ℂ f θ (Pi.single j 1) i

/-- Blueprint `bp:p2s2l1:toolkit`(a), vectors. -/
theorem L1Vec.norm_le {ι : Type*} [Fintype ι] {r A : ℝ} {f : (Fin n → ℂ) → ι → ℂ} (hr : 0 < r)
    (hf : DifferentiableOn ℂ f (complexStrip n r)) (hfp : IsPeriodicOn f r) (h : L1Vec r f A) :
    ∀ θ ∈ complexStrip n r, ‖f θ‖ ≤ A := sorry

/-- Blueprint `bp:p2s2l1:toolkit`(a), matrices (operator norm of the max norm = maximal row sum). -/
theorem L1Mat.norm_le {ι κ : Type*} [Fintype ι] [Fintype κ] {r A : ℝ} {M : (Fin n → ℂ) → Matrix ι κ ℂ}
    (hr : 0 < r) (hM : DifferentiableOn ℂ M (complexStrip n r)) (hMp : IsPeriodicOn M r) (h : L1Mat r M A) :
    ∀ θ ∈ complexStrip n r, ‖M θ‖ ≤ A := sorry

/-- Blueprint `bp:p2s2l1:toolkit`(b): the Wiener product, matrix times vector. -/
theorem L1Mat.mulVec {ι κ : Type*} [Fintype ι] [Fintype κ] {r A B : ℝ} {M : (Fin n → ℂ) → Matrix ι κ ℂ}
    {v : (Fin n → ℂ) → κ → ℂ} (hr : 0 < r) (hM : DifferentiableOn ℂ M (complexStrip n r)) (hMp : IsPeriodicOn M r)
    (hv : DifferentiableOn ℂ v (complexStrip n r)) (hvp : IsPeriodicOn v r) (hMA : L1Mat r M A)

-- MODULE KAM/Sharp2/L1/EntryB.lean

open KAM KAM.Torus KAM.Apost

/-- The ℓ¹ entry numbers of route B, per unit of the sup weighted error `μ₀`: `ci = Γ_∞(δ₀) γ δ₀^κ`, `cih = Γ_∞(δ₀/2)
γ (δ₀/2)^κ`, `cir = Γ_∞(ρ) γ ρ^κ`, `rN = ε¹_N/(γ δ₀^κ μ₀)`, `rL = ε¹_L/μ₀`, the ℓ¹ torsion cap `t1`, the stage-0 `e/μ₀`,
the certified `c_Th`, `c_DTh` of the design's (R1) (`2 M1`, `2 M2` by default), and whether the torsion cap is derived. -/
structure EntryNumB where
  ci : ℝ
  cih : ℝ
  cir : ℝ
  rN : ℝ
  rL : ℝ
  t1 : ℝ
  eμ : ℝ
  cTh : ℝ
  cDTh : ℝ
  derivedT : Bool
  tslack : ℝ

/-- Blueprint `bp:p2s2l1:constB`: the constants of `bp:p2s2:constantsB` at stage 0 with the ℓ¹ entry (each ingredient the
smaller of its sup and ℓ¹ bounds; the defect of `K` in `CΩK`, those of `K̃`, `K̄` in `CΩ`). Mirror: `stageB(..., l1=...)`. -/
def entryConstsB (p : StepIn) (l : EntryNumB) : StepConsts × ℝ :=
  let n : ℝ := p.n
  let g := p.γb * p.δ ^ p.κ
  let g1 := g * p.δ
  let sKt := (1 + p.θ) * p.sK
  let CΩ := 2 * (n - 1) * p.c1
  let CΩK := min CΩ (2 ^ (p.κ + 1) * n * (Real.exp 1)⁻¹ * l.cih * l.rN)
  let CEL := (p.sK + p.sN * p.q) / (1 - p.q ^ 2)
  let CEN := (p.sN + p.sK * p.q) / (1 - p.q ^ 2)
  let CE := min (p.lam * CEL + g * CEN) l.eμ
  let CDE := min (n * (p.sK + p.sN) * (max p.lam g + (p.sLt + p.sNt) * CE) / (1 - p.q)) (n * l.eμ)
  let CNN := n * (p.lam + p.sNt * CE)
  let avgB := min p.cbar (l.cir * l.rN) * (p.δ / p.r) ^ p.κ
  let C0 := if p.iso then p.sT * max (p.lam + p.t * avgB) (p.lamW + p.y0 * avgB) else p.sT * (p.lam + p.t * avgB)
  let CxiN := C0 + min p.c (l.ci * l.rN)
  let CxiN1 := C0 + l.ci * l.rN
  let c1N := min p.c1 (2 ^ (p.κ + 1) * l.cih * l.rN)
  let CKt := p.sN * CxiN
  let CDLt := n * p.sN * (C0 + c1N)
  let CDLtT := 2 * n * p.sN * (C0 + c1N)
  let CdN := (p.M1 * p.sK + n) * p.sB + p.sN * ((p.sLt * p.M1 + 2 * n) * p.sK + p.sLt * (p.M1 * p.sK + n)) * p.sB
  let CXN := p.M1 * p.sN + CdN
  let CEt := CE + CXN * CxiN + p.sN * g + p.M2 / 2 * CKt + (if p.iso then sKt * p.Θ * C0 else 0)
  let QtN := CDLtT * CEt + CNN * CxiN + p.δ * p.sLt * (p.M2 / 2) * CKt ^ 2
  let CgL := p.lam + p.t * CxiN + (if p.iso then p.Θ * C0 else 0)
  let CxiLs := p.c * CgL
  let CxiL := min CxiLs (l.ci * (l.rL + l.t1 * CxiN1 + (if p.iso then p.Θ * C0 else 0)))
  let CDK := sKt * CxiL + g * CKt
  let CEb := CEt + sKt * CgL
  let CG := n * p.sLt + 2 * n * p.sK
  let CBb := p.sB ^ 2 * CG * CDK / (1 - p.θ)
  let CNb := n * CDK * p.sB + p.sK * CBb
  let CNbT := CBb * p.sLt + 2 * n * CDK * p.sB
  let w := 2 * p.M1 * p.sLt * p.sK + 2 * n * p.sK + n * p.sLt
  let CDT0 := l.cTh * (CNbT * p.sN + p.sNt * CNb) + l.cDTh * p.sNt * p.sN * CDK * p.δ +
    g * (2 * p.sNt * p.sB * CDE + p.sB ^ 3 * w * (CΩ + CΩK))
  let CDT := if p.iso then max (CDT0 + p.Θ * C0 * g1) (n * p.Θ * C0 * g1 + 2 * n * g1) else CDT0
  let QbN := (1 + n * p.θξ) * (1 + p.θΩ) * QtN +
    (if p.iso then (1 + n * p.θξ) * p.Θ * CΩ * QtN * C0 * p.θξ / CxiLs else 0)
  let QL0 := p.sNt * (p.M2 / 2) * CKt ^ 2 * g1 + p.sB ^ 2 * CΩK * g ^ 2 + n * p.sNt * CEt * CxiL +
    p.sNt * CgL * (n * sKt * CxiL + g * CDLt) + CNbT * CEb
  let QL := if p.iso then max (QL0 + C0 * CgL * g1 + C0 * p.sNt * CDLt * p.Θ * g)
      (p.Mh2 / 2 * CKt ^ 2 * g1 + n * QtN * p.θξ * (p.Θ * CΩ + g1)) else QL0
  (⟨CΩ, CE, CDE, CNN, C0, CxiN, CKt, CDLt, CDLtT, CEt, QtN, CgL, CxiL, CDK, CEb, CG, CBb, CNb, CNbT, CDT, QbN, QL,
    max QL (2 ^ p.κ * QbN)⟩, CΩK)

/-- The entry step conditions (`bp:p2s2l1:constB`): those of `stepCond` with `CΩK` for the defect of `K`, the derived-
torsion condition when the torsion cap is derived, the iso composition condition with the sup `CξL`, and the local
contraction `2^{κ+1} Q X ≤ 1` (audit F8). -/
def entryCondB (p : StepIn) (l : EntryNumB) (X : ℝ) : Prop :=
  let k := (entryConstsB p l).1
  let CΩK := (entryConstsB p l).2
  let g := p.γb * p.δ ^ p.κ
  let w := 2 * p.M1 * p.sLt * p.sK + 2 * (p.n : ℝ) * p.sK + p.n * p.sLt
  max 1 (p.sB ^ 2) * CΩK * g * X ≤ p.q ∧ k.CDE * g * X ≤ p.n ∧ k.CxiL * X ≤ p.θξ ∧
    k.CΩ * k.CgL * X ≤ p.θΩ ∧ k.CKt * (g * p.δ) * X ≤ 1 ∧ k.CDLt * g * X ≤ p.θ * p.sK ∧
    k.CDLtT * g * X ≤ p.θ * p.sLt ∧ p.sB * k.CG * k.CDK * X ≤ p.θ ∧ p.sT * k.CDT * X ≤ p.θ ∧
    (l.derivedT = true → (p.sNt * p.sB * k.CDE + p.sB ^ 3 * w * CΩK) * g * X ≤ l.tslack) ∧
    (p.iso = true → p.c * k.CgL * X ≤ p.θξ) ∧
    2 ^ (p.κ + 1) * k.Q * X ≤ 1

/-- The entry numbers from the data (`bp:p2s2l1:sec-B`): `p = max {σ_{Nᵀ}, σ_{Lᵀ}/(γ δ₀^κ), 1 (iso)}`, `μ₀ = p e`,
`λ₀ = σ_{Nᵀ}/p` (audit AUDIT-SHARP-V2-iso-1, F4). -/
def entryNumB (d : L1Data) (κ γ γb ρ δ₀ sLt sNt pe cTh cDTh tslack : ℝ) (derivedT : Bool) : EntryNumB :=
  { ci := d.G0 * γb * δ₀ ^ κ, cih := d.G0h * γb * (δ₀ / 2) ^ κ, cir := d.Gρ * γb * ρ ^ κ,
    rN := d.aN * sLt / (γ * δ₀ ^ κ * pe), rL := d.aL * sNt / pe, t1 := d.t1, eμ := 1 / pe,
    cTh := cTh, cDTh := cDTh, derivedT := derivedT, tslack := tslack }

/-- Blueprint `bp:p2s2l1:entryB`: the modified step at stage 0 with the ℓ¹ entry. Hypotheses: the standing frozen
hypotheses (periodicity and reality of `h`, audit F4), the caps of the step, the refined bounds, the sup small-divisor
constants at the stage-0 bites for the actual `ω` (F3, F9), the entry data, and the stage-0 weighted error `μ₀` with
the weight `0 < λ ≤ 1` (iso-F4) dominating the three error components; the entry numbers `l` are upper bounds of their
definitions (`ci ≥ Γ_∞(δ₀) γ δ₀^κ`, ..., `rN μ₀ γ δ₀^κ ≥ ε¹_N`, `rL μ₀ ≥ ε¹_L`, `eμ μ₀ ≥ e`). Conclusions: those of
`modified_step` with `entryConstsB`. -/
theorem entry_step_B {n : ℕ} (p : StepIn) (l : EntryNumB) (d : L1Data) {τ M σK σB σT γ β e μ₀ : ℝ}
    {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))} {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ}
    {K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ω : Fin n → ℝ}
    (hpn : p.n = n) (hS : Standing n τ M σK σB σT 𝔅 h ω γ) (hγb : γ ≤ p.γb) (hδ : 0 < p.δ) (hδ1 : p.δ ≤ 1)
    (hr : 3 * p.δ < p.r) (hκ : p.κ = τ) (hτ1 : 1 ≤ τ)
    (hc : SDConst n p.κ γ p.δ p.c {ω}) (hc1 : SDConst n p.κ γ (p.δ / 2) (p.c1 / 2 ^ (p.κ + 1)) {ω})
    (hcbar : SDConst n p.κ γ p.r p.cbar {ω})
    (hd : d.WF) (hl : L1Hyp d h K ω p.r p.δ e p.sLt p.sNt)
    (hci : d.G0 * γ * p.δ ^ p.κ ≤ l.ci) (hcih : d.G0h * γ * (p.δ / 2) ^ p.κ ≤ l.cih)
    (hcir : d.Gρ * γ * p.r ^ p.κ ≤ l.cir) (ht1 : d.t1 ≤ l.t1)
    (hμ₀ : 0 < μ₀) (hlam : 0 < p.lam ∧ p.lam ≤ 1) (hlamW : p.iso = true → 0 < p.lamW ∧ p.lamW ≤ 1)
    (hμL : p.sNt * e ≤ p.lam * μ₀) (hμN : p.sLt * e ≤ γ * p.δ ^ p.κ * μ₀)
    (hμW : p.iso = true → e ≤ p.lamW * μ₀) (heμ : e ≤ l.eμ * μ₀)
    (hrN : d.aN * p.sLt * e ≤ l.rN * (γ * p.δ ^ p.κ * μ₀)) (hrL : d.aL * p.sNt * e ≤ l.rL * μ₀)
    (hTh : ∀ z ∈ 𝔅, ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ l.cTh / 2 ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ l.cDTh / 2)
    (hK : DifferentiableOn ℂ K (complexStrip n p.r)) (hKp : IsPrimaryOnStrip K p.r) (hKr : IsRealOnReals K)
    (hball : ∀ θ ∈ complexStrip n p.r, Metric.ball (K θ) β ⊆ 𝔅)
    (hdX : ∀ z ∈ 𝔅, ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ p.M1)
    (hd2X : ∀ z ∈ 𝔅, ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ p.M2)
    (hL : ∀ θ ∈ complexStrip n p.r, ‖tangentMat K θ‖ ≤ p.sK)
    (hLt : ∀ θ ∈ complexStrip n p.r, ‖(tangentMat K θ)ᵀ‖ ≤ p.sLt)
    (hG : ∀ θ ∈ complexStrip n p.r, IsUnit (gramMatrix K θ).det ∧ ‖(gramMatrix K θ)⁻¹‖ ≤ p.sB)
    (hN : ∀ θ ∈ complexStrip n p.r, ‖normalMat K θ‖ ≤ p.sN)
    (hNt : ∀ θ ∈ complexStrip n p.r, ‖(normalMat K θ)ᵀ‖ ≤ p.sNt)
    (hT : ∀ θ ∈ complexStrip n (p.r - p.δ), ‖torsion h K ω θ‖ ≤ p.t)
    (hTa : IsUnit (torsionAverage h K ω).det ∧ ‖(torsionAverage h K ω)⁻¹‖ ≤ p.sT)
    (hE : ∀ θ ∈ complexStrip n p.r, ‖invarianceError h K ω θ‖ ≤ e)
    (hcond : entryCondB p l (μ₀ / (γ * p.δ ^ (p.κ + 1))))
    (hβ : (entryConstsB p l).1.CDK * μ₀ / (γ * p.δ ^ p.κ) < β) :
    let X := μ₀ / (γ * p.δ ^ (p.κ + 1))
    let Kb := modNewton h K ω (p.r - 2 * p.δ)
    let k := (entryConstsB p l).1
    DifferentiableOn ℂ Kb (complexStrip n (p.r - 2 * p.δ)) ∧ IsPrimaryOnStrip Kb (p.r - 2 * p.δ) ∧
      IsRealOnReals Kb ∧
      (∀ θ ∈ complexStrip n (p.r - 2 * p.δ), ‖Kb θ - K θ‖ ≤ k.CDK * μ₀ / (γ * p.δ ^ p.κ)) ∧
      (∀ θ ∈ complexStrip n (p.r - 3 * p.δ), ‖etaN h Kb ω θ‖ ≤ k.QbN * μ₀ ^ 2 / p.δ) ∧
      (∀ θ ∈ complexStrip n (p.r - 3 * p.δ), ‖etaL h Kb ω θ‖ ≤ k.QL * μ₀ ^ 2 / (γ * p.δ ^ (p.κ + 1))) ∧
      (∀ θ ∈ complexStrip n (p.r - 3 * p.δ), ‖fderiv ℂ Kb θ - fderiv ℂ K θ‖ ≤ p.n * k.CDK * X) ∧
      (∀ θ ∈ complexStrip n (p.r - 3 * p.δ), IsUnit (gramMatrix Kb θ).det ∧
        ‖(gramMatrix Kb θ)⁻¹ - (gramMatrix K θ)⁻¹‖ ≤ k.CBb * X) ∧
      (∀ θ ∈ complexStrip n (p.r - 3 * p.δ), ‖normalMat Kb θ - normalMat K θ‖ ≤ k.CNb * X ∧
        ‖(normalMat Kb θ)ᵀ - (normalMat K θ)ᵀ‖ ≤ k.CNbT * X) ∧
      (∀ θ ∈ complexStrip n (p.r - 7 * p.δ / 2), ‖torsion h Kb ω θ - torsion h K ω θ‖ ≤ k.CDT * X) ∧
      IsUnit (torsionAverage h Kb ω).det ∧
      ‖(torsionAverage h Kb ω)⁻¹ - (torsionAverage h K ω)⁻¹‖ ≤ p.sT ^ 2 * k.CDT * X / (1 - p.θ) := sorry

end KAM.Sharp2.L1

/-! ============================================================================================
## Part 5. Route A: stage constants, the entry, the embedding into the B5 engine
(`bp:p2s2l1:constA`, `bp:p2s2l1:frame`, `bp:p2s2l1:entryA`, `bp:p2s2l1:engine`(b))

-- MODULE KAM/Sharp2/L1/RouteA.lean  (the route-A stage constants are joint with SHARP2-B9; the B10 delta is `EntryNumA`)
namespace KAM.Sharp2.L1

open KAM KAM.Torus KAM.Apost

/-- Inputs of one route-A stage with the levers of `bp:p2s2:constantsA` (refined data, torsion cap, `Ω_K` through `ηᴺ`,
full-bite average, δ powers kept unless `keep = false` (tail), caps for the barred factors): mirror `stageA` with options
`L1, torsion_cap, omega_eta, avg_full_bite, keep_delta, caps_tilde`, `η = 1`, `q_S = 1/2`. -/
structure StepInA where
  n : ℕ
  κ : ℝ
  M1 : ℝ
  M2 : ℝ
  Mh1 : ℝ
  Mh2 : ℝ
  sK : ℝ
  sLt : ℝ
  sB : ℝ
  sN : ℝ
  sNt : ℝ
  sT : ℝ
  t : ℝ
  t0 : ℝ
  c : ℝ
  ch : ℝ
  cr : ℝ
  θ : ℝ
  γb : ℝ
  δ : ℝ
  r : ℝ
  keep : Bool := true
  iso : Bool := false
  Θ : ℝ := 0

/-- The ℓ¹ entry numbers of route A (per unit of the sup data): `ci = Γ_∞(δ₀) γ̄ δ₀^κ`, `cir = Γ_∞(ρ) γ̄ ρ^κ`, the ratios
`rN = aN`, `rL = aL`, the ℓ¹ torsion cap, the ℓ¹ frame caps; `entry = false` gives the sup stage, `frame` the ℓ¹ frame
increment. -/
structure EntryNumA where
  ci : ℝ
  cir : ℝ
  rN : ℝ
  rL : ℝ
  t1 : ℝ
  nuK : ℝ
  nuN : ℝ
  entry : Bool
  frame : Bool

/-- The sup stage (no entry). -/
def EntryNumA.sup : EntryNumA := ⟨0, 0, 0, 0, 0, 0, 0, false, false⟩

/-- The constants of one route-A stage. -/
structure StageConstsA where
  CE : ℝ
  CDK : ℝ
  CDw : ℝ
  Csym : ℝ
  CG : ℝ
  CB : ℝ
  CNd : ℝ
  CNtd : ℝ
  CDT : ℝ
  rhoT : ℝ
  kc1 : ℝ
  kc2 : ℝ

/-- Route-A stage constants with the optional ℓ¹ entry (`bp:p2s2l1:constA`); mirror `stageA(..., l1=...)`. -/
def stageConstsA (q : StepInA) (l : EntryNumA) : StageConstsA :=
  let n : ℝ := q.n
  let a := 2 * q.κ
  let G : ℕ → ℝ → ℝ := fun i j => q.γb ^ i * (if q.keep then q.δ ^ j else 1)
  let qS : ℝ := 1 / 2
  let COm := 2 * (n - 1) * q.c * q.sLt
  let COmK := if l.entry then min COm (n * (Real.exp 1)⁻¹ * l.ci * l.rN * q.sLt) else COm
  let Csym := max 1 (q.sB ^ 2) * COmK
  let Cred := n * max q.sNt (q.sLt + 2 * q.sN)
  let w := 2 * q.M1 * q.sLt * q.sK + (2 * n * q.sK + n * q.sLt)
  let cavg0 := (if q.keep then q.cr * (q.δ / q.r) ^ q.κ else q.c)
  let cavg := if l.entry then min cavg0 (l.cir * (q.δ / q.r) ^ q.κ * l.rN) else cavg0
  let cN1 := if l.entry then l.ci * l.rN else q.c
  let cN := min q.c cN1
  let HY := q.Mh1 * q.sN
  let CS := if q.iso then q.sT * max (q.sNt * G 1 q.κ + q.t0 * cavg * q.sLt) (G 1 q.κ + cavg * HY * q.sLt)
    else q.sT * (q.sNt * G 1 q.κ + q.t0 * cavg * q.sLt)
  let CxiN := CS + cN * q.sLt
  let CxiN1 := CS + cN1 * q.sLt
  let Cg := q.t * CxiN + (if q.iso then q.Θ * CS else 0) + q.sNt * G 1 q.κ
  let Cdxi := max Cg (q.sLt * G 1 q.κ)
  let CxiL1 := if l.entry then l.ci * (l.t1 * CxiN1 + l.rL * q.sNt * G 1 q.κ + (if q.iso then q.Θ * CS else 0))
    else q.c * Cg
  let CxiL := min (q.c * Cg) CxiL1
  let Cxi := max CxiL (G 1 q.κ * CxiN)
  let CDK := q.sK * CxiL + q.sN * G 1 q.κ * CxiN
  let CDK1 := l.nuK * CxiL1 + l.nuN * G 1 q.κ * CxiN1
  let kc1 := if l.entry ∧ l.frame then min n ((Real.exp 1)⁻¹ * CDK1 / CDK) else n
  let kc2 := if l.entry ∧ l.frame then min (2 * n) (2 * n * (Real.exp 1)⁻¹ * CDK1 / CDK) else 2 * n
  let P := q.sK + q.sN
  let Clin0 := Cred * Cxi + Csym * Cdxi
  let CDw := q.Θ * G 1 q.κ * CS
  let CP := (q.M1 * q.sK + n) * (1 + q.sB) +
    q.sN * ((q.sLt * q.M1 + 2 * n) * q.sK + q.sLt * (q.M1 * q.sK + n)) * q.sB
  let CdK := CP * Cxi + P * G 1 q.κ * Cdxi
  let CE := if q.iso then
      max (P / (1 - qS) * G 2 (a - 1) * (Clin0 + q.Θ * COmK * CS) + 1 / 2 * q.M2 * CDK ^ 2 + G 1 q.κ * CS * CdK)
        (G 1 (q.κ - 1) * n * (q.sLt * G 1 q.κ + q.Θ * COmK) * CxiL + 1 / 2 * q.Mh2 * CDK ^ 2)
    else P / (1 - qS) * G 2 (a - 1) * Clin0 + 1 / 2 * q.M2 * CDK ^ 2
  let CG := kc1 * q.sLt + kc2 * q.sK
  let CB := q.sB ^ 2 * CG / (1 - q.θ)
  let CNd := kc1 * q.sB + q.sK * CB
  let CNtd := CB * q.sLt + kc2 * q.sB
  let CDab := 2 * q.M1 * (CNtd * q.sN + q.sNt * CNd) + 2 * q.M2 * q.sNt * q.sN * (if q.keep then q.δ else 1)
  let CDcd := q.sB ^ 3 * w * (kc2 * q.sK + kc1 * q.sLt)
  let CDT0 := (CDab + CDcd) * CDK + n * q.sNt * q.sB * 2 * G 2 a + q.sB ^ 3 * w * (COm + COmK) * G 1 q.κ
  let D1 := if q.keep then q.δ else 1
  let CDT := if q.iso then max (CDT0 + CDw * D1) (n * CDw * D1 + n * 2 * q.sNt * G 2 a) else CDT0
  ⟨CE, CDK, CDw, Csym, CG, CB, CNd, CNtd, CDT, q.sT ^ 2 / (1 - q.θ) * CDT, kc1, kc2⟩

/-- Route-A step conditions at `x = ε/(γ⁴ δ^{4κ})` (mirror of `stageA`'s list) plus the local contraction
`2^{4κ} C_E x ≤ 1`. -/
def stageCondA (q : StepInA) (l : EntryNumA) (x : ℝ) : Prop :=
  let k := stageConstsA q l
  let a := 2 * q.κ
  let G : ℕ → ℝ → ℝ := fun i j => q.γb ^ i * (if q.keep then q.δ ^ j else 1)
  G 4 (2 * a - 1) * x ≤ 1 ∧ k.Csym * G 3 (3 * q.κ - 1) * x ≤ 1 / 2 ∧ k.CE * x ≤ 1 ∧
    q.sT * k.CDT * G 2 (a - 1) * x ≤ q.θ ∧ q.sB * k.CG * k.CDK * G 2 (a - 1) * x ≤ q.θ ∧
    2 ^ (4 * q.κ) * k.CE * x ≤ 1

/-- The caps of route A: 0 `|DK|`, 1 `|B|`, 2 `|⟨T⟩⁻¹|`, 3 `|Lᵀ|`, 4 `|N|`, 5 `|Nᵀ|`, 6 `|T|` (torsion cap), 7 frequency. -/
abbrev capsA : ℕ := 8

/-- Blueprint `bp:p2s2l1:engine`(b): route A in the route-B-normalised engine, `X = x`, `μ = ε/(γ³ δ^{3κ-1})`,
`Q = 2^{3κ-1} C_E` (so `2^{κ+1} Q = 2^{4κ} C_E`, audit F5), displacement `γ² δ^{2κ-1} C_ΔK`, frame weights `γ² δ^{2κ-1}`,
frequency weight `γ² δ^{2κ}`. -/
def engineConstsA (q : StepInA) (l : EntryNumA) (γ : ℝ) : StageConsts capsA :=
  let k := stageConstsA q l
  let wF := γ ^ 2 * q.δ ^ (2 * q.κ - 1)
  { Q := 2 ^ (3 * q.κ - 1) * k.CE
    cond := stageCondA q l
    CDK := wF * k.CDK
    Cp := γ * q.δ ^ (q.κ - 1) * k.CDw
    rate := ![k.kc1 * k.CDK, k.CB * k.CDK, k.rhoT, k.kc2 * k.CDK, k.CNd * k.CDK, k.CNtd * k.CDK, k.CDT, k.CDw]
    wt := ![wF, wF, wF, wF, wF, wF, wF, γ ^ 2 * q.δ ^ (2 * q.κ)] }

/-- The entry numbers of route A from the data. -/
def entryNumA (d : L1Data) (κ γb ρ δ₀ : ℝ) (frame : Bool) : EntryNumA :=
  ⟨d.G0 * γb * δ₀ ^ κ, d.Gρ * γb * ρ ^ κ, d.aN, d.aL, d.t1, d.nuK, d.nuN, true, frame⟩

/-- Blueprint `bp:p2s2l1:frame`: the ℓ¹ frame increment (route A). -/
theorem norm_fderiv_newtonMap_sub_le_l1 {n : ℕ} {τ M σK σB σT γ r δ εK : ℝ}
    {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))} {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ} {ω : Fin n → ℝ}
    {K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} (hS : Standing n τ M σK σB σT 𝔅 h ω γ) (hδ : 0 < δ)
    (h3δ : 3 * δ < r) (hK : DifferentiableOn ℂ K (complexStrip n r))
    (hΔ : L1Vec (r - 2 * δ) (fun θ => sumVec n (correction h K ω θ)) εK) :
    ∀ θ ∈ complexStrip n (r - 3 * δ),
      ‖fderiv ℂ (newtonMap h K ω (r - 2 * δ)) θ - fderiv ℂ K θ‖ ≤ εK / (Real.exp 1 * δ) ∧
      ‖(tangentMat (newtonMap h K ω (r - 2 * δ)) θ - tangentMat K θ)ᵀ‖ ≤ 2 * n * εK / (Real.exp 1 * δ) := sorry

/-- The admissible states of route A in the engine (`bp:p2s2l1:engine`(b)): `K` on `T_r`, ball `β`, error
`ε ≤ γ³ b(r)^{3κ-1} μ` with the schedule bite `b(r) = (r - ρ + 6δ₀)/6`, caps `m`. -/
structure AdmA {n : ℕ} (τ ρ ρinf γ : ℝ) (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))) (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (ω : Fin n → ℝ) (r β μ : ℝ) (m : Fin capsA → ℝ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) : Prop where
  holo : DifferentiableOn ℂ K (complexStrip n r)
  primary : IsPrimaryOnStrip K r
  real : IsRealOnReals K
  ball : ∀ θ ∈ complexStrip n r, Metric.ball (K θ) β ⊆ 𝔅
  error : ∀ θ ∈ complexStrip n r, ‖invarianceError h K ω θ‖ ≤
    γ ^ 3 * ((r - ρ + 6 * bite0 ρ ρinf) / 6) ^ (3 * τ - 1) * μ
  capDK : ∀ θ ∈ complexStrip n r, ‖fderiv ℂ K θ‖ ≤ m 0
  capB : ∀ θ ∈ complexStrip n r, IsUnit (gramMatrix K θ).det ∧ ‖(gramMatrix K θ)⁻¹‖ ≤ m 1
  capTinv : IsUnit (torsionAverage h K ω).det ∧ ‖(torsionAverage h K ω)⁻¹‖ ≤ m 2
  capLt : ∀ θ ∈ complexStrip n r, ‖(tangentMat K θ)ᵀ‖ ≤ m 3
  capN : ∀ θ ∈ complexStrip n r, ‖normalMat K θ‖ ≤ m 4
  capNt : ∀ θ ∈ complexStrip n r, ‖(normalMat K θ)ᵀ‖ ≤ m 5
  capT : ∀ θ ∈ complexStrip n (r - (r - ρ + 6 * bite0 ρ ρinf) / 6), ‖torsion h K ω θ‖ ≤ m 6

/-- Blueprint `bp:p2s2l1:entryA`: the route-A step at stage 0 with the ℓ¹ entry, ending in an engine state at
`ρ₁ = ρ - 3δ₀`. -/
theorem entry_step_A {n : ℕ} (q : StepInA) (d : L1Data) (frame : Bool) {τ M σK σB σT γ ρinf β e : ℝ}
    {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))} {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ}
    {K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)} {ω : Fin n → ℝ} {m₀ : Fin capsA → ℝ}
    (hqn : q.n = n) (hS : Standing n τ M σK σB σT 𝔅 h ω γ) (hγb : γ ≤ q.γb) (hκ : q.κ = τ) (hτ1 : 1 ≤ τ)
    (hρinf : 0 < ρinf) (hρ : ρinf < q.r) (hδ : q.δ = bite0 q.r ρinf)
    (hc : SDConst n τ γ q.δ q.c {ω}) (hch : SDConst n τ γ (q.δ / 2) q.ch {ω}) (hcr : SDConst n τ γ q.r q.cr {ω})
    (hd : d.WF) (hl : L1Hyp d h K ω q.r q.δ e q.sLt q.sNt)
    (hK : AdmA τ q.r ρinf γ 𝔅 h ω q.r β (e / (γ ^ 3 * q.δ ^ (3 * τ - 1))) m₀ K)
    (hM : ∀ z ∈ 𝔅, ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ q.M1 ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ q.M2)
    (hm : ∀ i, m₀ i ≤ ![q.sK, q.sB, q.sT, q.sLt, q.sN, q.sNt, q.t, 0] i) (htR : d.tR ≤ q.t0) (ht0 : q.t0 ≤ q.t)
    (hcond : stageCondA q (entryNumA d τ q.γb q.r q.δ frame) (e / (γ ^ 4 * q.δ ^ (4 * τ))))
    (hβ : (engineConstsA q (entryNumA d τ q.γb q.r q.δ frame) γ).CDK * (e / (γ ^ 3 * q.δ ^ (3 * τ - 1))) /
      (γ * q.δ ^ τ) < β) :
    let C := engineConstsA q (entryNumA d τ q.γb q.r q.δ frame) γ
    let μ := e / (γ ^ 3 * q.δ ^ (3 * τ - 1))
    AdmA τ q.r ρinf γ 𝔅 h ω (q.r - 3 * q.δ) (β - C.CDK * μ / (γ * q.δ ^ τ)) (C.Q * μ ^ 2 / (γ * q.δ ^ (τ + 1)))
      (fun i => m₀ i + C.rate i * C.wt i * (μ / (γ * q.δ ^ (τ + 1)))) (newtonMap h K ω (q.r - 2 * q.δ)) ∧
    ∀ θ ∈ complexStrip n (q.r - 3 * q.δ), ‖newtonMap h K ω (q.r - 2 * q.δ) θ - K θ‖ ≤ C.CDK * μ / (γ * q.δ ^ τ) :=
  sorry

/-- The route-A scheme for the engine (states: tori; step: `newtonMap` at the strip `r - 2δ`). -/
def routeAScheme {n : ℕ} (τ ρ ρinf γ : ℝ) (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))) (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ)
    (ω : Fin n → ℝ) : StageScheme n capsA ((Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) where
  tor := id
  par := fun _ => 0
  Adm := fun r β μ m K => AdmA τ ρ ρinf γ 𝔅 h ω r β μ m K
  step := fun r δ K => newtonMap h K ω (r - 2 * δ)

/-- The schedule index of a bite: `s` with `δ = δ₀/2^s` on the schedule. -/
def schedIdx (ρ ρinf δ : ℝ) : ℕ := Nat.log 2 (Nat.floor (bite0 ρ ρinf / δ))

/-- The route-A stage inputs at a bite `δ` (the function of the bite required by `StageScheme.IsNewton`): the sup
constants `cS (schedIdx δ)` on the prefix, the uniform tail constant `ctail` (with absorbed powers) for `δ ≤ δ_kpre`. -/
def stageInAt (base : StepInA) (cS : ℕ → ℝ × ℝ × ℝ) (ctail ρ ρinf : ℝ) (kpre : ℕ) (δ : ℝ) : StepInA :=
  let tail := decide (δ ≤ biteS ρ ρinf kpre)
  let c3 := if tail then (ctail, ctail, ctail) else cS (schedIdx ρ ρinf δ)
  { base with c := c3.1, ch := c3.2.1, cr := c3.2.2, δ := δ, r := ρ - 6 * bite0 ρ ρinf + 6 * δ, keep := !tail }

/-- The sup route-A step as an engine instance from stage 1 (owned by SHARP2-B9; consumed here). -/
theorem routeA_isNewton {n : ℕ} (base : StepInA) (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ)
    {τ M σK σB σT γ ρ ρinf : ℝ} {𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ))} {h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ}
    {ω : Fin n → ℝ} (σcap : Fin capsA → ℝ) (hS : Standing n τ M σK σB σT 𝔅 h ω γ) (hγb : γ ≤ base.γb)
    (hcS : ∀ s, 1 ≤ s → s < kpre → SDConst n τ γ (biteS ρ ρinf s) (cS s).1 {ω} ∧
      SDConst n τ γ (biteS ρ ρinf s / 2) (cS s).2.1 {ω} ∧ SDConst n τ γ (stripS ρ ρinf s) (cS s).2.2 {ω})
    (hct : ∀ δ, 0 < δ → δ ≤ biteS ρ ρinf kpre → SDConst n τ γ δ ctail {ω}) :
    (routeAScheme τ ρ ρinf γ 𝔅 h ω).IsNewton τ γ (biteS ρ ρinf 1)
      (fun δ => engineConstsA (stageInAt base cS ctail ρ ρinf kpre δ) EntryNumA.sup γ) σcap := sorry

end KAM.Sharp2.L1

/-- The hypotheses H1–H5 of the frozen G1 body (verbatim split of `KAM.Sharp.G1Body`). -/
def G1Hyps (n : ℕ) (τ ρ M σK σB σT μ γbar : ℝ) (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ)))
    (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω : Fin n → ℝ)
    (γ e : ℝ) : Prop :=
  IsOpen 𝔅 ∧
    (∀ z ∈ 𝔅, ∀ k : Fin n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ 𝔅) ∧
    DifferentiableOn ℂ h 𝔅 ∧
    (∀ z ∈ 𝔅, ∀ k : Fin n → ℤ, h (z.1 + (fun i => (k i : ℂ)), z.2) = h z) ∧
    (∀ θ I : Fin n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ 𝔅 →
      (h ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) ∧
    (∀ z ∈ 𝔅, ‖fderiv ℂ h z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ h z‖ ≤ M ∧
      ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ M ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ M) ∧
    DifferentiableOn ℂ K (complexStrip n ρ) ∧ IsPrimaryOnStrip K ρ ∧ IsRealOnReals K ∧
    (∀ θ ∈ complexStrip n ρ, Metric.ball (K θ) μ ⊆ 𝔅) ∧
    (∀ θ ∈ complexStrip n ρ, ‖fderiv ℂ K θ‖ ≤ σK - μ) ∧
    (∀ θ ∈ complexStrip n ρ, IsUnit (gramMatrix K θ).det ∧
      ∀ i, ∑ j, ‖(gramMatrix K θ)⁻¹ i j‖ ≤ σB - μ) ∧
    IsUnit (torsionAverage h K ω).det ∧
    (∀ i, ∑ j, ‖(torsionAverage h K ω)⁻¹ i j‖ ≤ σT - μ) ∧
    0 < γ ∧ γ ≤ γbar ∧ IsDiophantine γ τ ω ∧
    (∀ θ ∈ complexStrip n ρ, ‖invarianceError h K ω θ‖ ≤ e)

/-- The conclusions of the frozen G1 body (verbatim split of `KAM.Sharp.G1Body`). -/
def G1Concl (n : ℕ) (τ ρ ρinf σK μ : ℝ) (𝔅 : Set ((Fin n → ℂ) × (Fin n → ℂ)))
    (h : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (ω : Fin n → ℝ)
    (γ e C₂ : ℝ) : Prop :=
  ∃ Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ),
    DifferentiableOn ℂ Kinf (complexStrip n ρinf) ∧ IsPrimaryOnStrip Kinf ρinf ∧
    IsRealOnReals Kinf ∧
    (∀ θ ∈ complexStrip n ρinf, invarianceError h Kinf ω θ = 0) ∧
    (∀ θ ∈ complexStrip n ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ 𝔅) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
    ∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ C₂ * e / (γ ^ 2 * ρ ^ (2 * τ))

/-- The split is the frozen body (definitional; proof `Iff.rfl` after unfolding the conjunctions). -/
theorem g1Body_iff_split (n : ℕ) (τ ρ ρinf M σK σB σT μ γbar C₁ C₂ : ℝ) :
    KAM.Sharp.G1Body n τ ρ ρinf M σK σB σT μ γbar C₁ C₂ ↔
      ∀ 𝔅 h K ω γ e, G1Hyps n τ ρ M σK σB σT μ γbar 𝔅 h K ω γ e →
        C₁ * e / (γ ^ 4 * ρ ^ (4 * τ)) < 1 → G1Concl n τ ρ ρinf σK μ 𝔅 h K ω γ e C₂ := by
  constructor
  · intro H 𝔅 h K ω γ e ⟨h1, h2, h3, h4, h5, h6, h7, h8, h9, h10, h11, h12, h13, h14, h15, h16, h17, h18⟩ hs
    exact H 𝔅 h K ω γ e h1 h2 h3 h4 h5 h6 h7 h8 h9 h10 h11 h12 h13 h14 h15 h16 h17 h18 hs
  · intro H 𝔅 h K ω γ e h1 h2 h3 h4 h5 h6 h7 h8 h9 h10 h11 h12 h13 h14 h15 h16 h17 h18 hs
    exact H 𝔅 h K ω γ e ⟨h1, h2, h3, h4, h5, h6, h7, h8, h9, h10, h11, h12, h13, h14, h15, h16, h17, h18⟩ hs

/-- The refined body with the ℓ¹ entry: the frozen hypotheses, the refined data (R1)–(R3) (torsion cap `t` on
`T_{ρ-δ₀}`), the ℓ¹ entry data (L1)–(L4), the sup small-divisor premises at every prefix bite and in the tail for the
quantified `ω` (L5), and the smallness, imply the frozen conclusion. -/
def G1BodyL1 (n : ℕ) (τ ρ ρinf M σK σB σT μ γbar : ℝ) (rd : RefData) (d : L1Data) (t : ℝ)
    (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ) (C₁ C₂ : ℝ) : Prop :=
  ∀ 𝔅 h K ω γ e, G1Hyps n τ ρ M σK σB σT μ γbar 𝔅 h K ω γ e → RefHyp rd 𝔅 h K ρ →
    (∀ θ ∈ complexStrip n (ρ - bite0 ρ ρinf), ‖torsion h K ω θ‖ ≤ t) →
    L1Hyp d h K ω ρ (bite0 ρ ρinf) e rd.sLt rd.sNt →
    (∀ s < kpre, SDConst n τ γ (biteS ρ ρinf s) (cS s).1 {ω} ∧
      SDConst n τ γ (biteS ρ ρinf s / 2) (cS s).2.1 {ω} ∧ SDConst n τ γ (stripS ρ ρinf s) (cS s).2.2 {ω}) →
    (∀ δ, 0 < δ → δ ≤ biteS ρ ρinf kpre → SDConst n τ γ δ ctail {ω}) →
    C₁ * e / (γ ^ 4 * ρ ^ (4 * τ)) < 1 → G1Concl n τ ρ ρinf σK μ 𝔅 h K ω γ e C₂

/-- The stage inputs of route A for the standing data: stage `s` at the schedule bite with the sup constants `cS s`. -/
def stageInA (n : ℕ) (τ ρ ρinf : ℝ) (rd : RefData) (Mh1 Mh2 sK sB sT t γb : ℝ) (cS : ℕ → ℝ × ℝ × ℝ) (s : ℕ) :
    StepInA :=
  { n := n, κ := τ, M1 := rd.M1, M2 := rd.M2, Mh1 := Mh1, Mh2 := Mh2, sK := sK, sLt := rd.sLt, sB := sB,
    sN := rd.sN, sNt := rd.sNt, sT := sT, t := t, t0 := t, c := (cS s).1, ch := (cS s).2.1, cr := (cS s).2.2,
    θ := 1 / 8, γb := γb, δ := biteS ρ ρinf s, r := stripS ρ ρinf s }

/-- `X⋆` certifies route A with the ℓ¹ entry: the entry conditions at `X⋆`, the prefix conditions, the tail
conditions on `[0, X_k]`, the tail decay `≤ 1/2`, every cap sum (initial value plus entry plus engine increments,
checked at `γ̄`), and the strict ball condition. Caps `σcap`, initial values `m₀` (audit F3). -/
def routeAOkL1 (n : ℕ) (τ ρ ρinf : ℝ) (rd : RefData) (Mh1 Mh2 sK sB sT t γb : ℝ) (d : L1Data) (frame : Bool)
    (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ) (σcap m₀ : Fin capsA → ℝ) (β₀ Xstar : ℝ) : Prop :=
  let P := stageInA n τ ρ ρinf rd Mh1 Mh2 sK sB sT t γb cS
  let C : ℕ → StageConsts capsA := fun s =>
    engineConstsA (P s) (if s = 0 then entryNumA d τ γb ρ (bite0 ρ ρinf) frame else EntryNumA.sup) γb
  let Ct : StageConsts capsA := engineConstsA { P kpre with c := ctail, ch := ctail, cr := ctail, keep := false }
    EntryNumA.sup γb
  let X := xSeq τ (fun s => (C s).Q) Xstar
  1 ≤ kpre ∧ (∀ s < kpre, (C s).cond (X s)) ∧ (∀ x, 0 ≤ x → x ≤ X kpre → Ct.cond x) ∧
    2 ^ (τ + 1) * Ct.Q * X kpre ≤ 1 / 2 ∧
    (∀ i, m₀ i + (∑ s ∈ Finset.range kpre, (C s).rate i * (C s).wt i * X s) + 2 * Ct.rate i * Ct.wt i * X kpre
      ≤ σcap i) ∧
    (∑ s ∈ Finset.range kpre, (C s).CDK * biteS ρ ρinf s * X s) + 2 * Ct.CDK * biteS ρ ρinf kpre * X kpre < β₀

/-- The route-A ball sum without `γ²` (`bp:p2s2l1:engine`(c)): `S̃ = Σ C_ΔK,s δ_s^{2κ} X_s` (+ tail). -/
def ballSumA (n : ℕ) (τ ρ ρinf : ℝ) (rd : RefData) (Mh1 Mh2 sK sB sT t γb : ℝ) (d : L1Data) (frame : Bool)
    (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ) (Xstar : ℝ) : ℝ :=
  let P := stageInA n τ ρ ρinf rd Mh1 Mh2 sK sB sT t γb cS
  let k : ℕ → StageConstsA := fun s =>
    stageConstsA (P s) (if s = 0 then entryNumA d τ γb ρ (bite0 ρ ρinf) frame else EntryNumA.sup)
  let kt := stageConstsA { P kpre with c := ctail, ch := ctail, cr := ctail, keep := false } EntryNumA.sup
  let X := xSeq τ (fun s => 2 ^ (3 * τ - 1) * (k s).CE) Xstar
  (∑ s ∈ Finset.range kpre, (k s).CDK * biteS ρ ρinf s ^ (2 * τ) * X s) +
    2 * kt.CDK * biteS ρ ρinf kpre ^ (2 * τ) * X kpre

/-- Blueprint `bp:p2s2l1:g1A` / `bp:p2s2l1:frozen` (route A, the PlasmaLean route): `C₁ = ρ^{4τ}/(X⋆ δ₀^{4τ})`,
`C₂ = ρ^{2τ} S̃/(X⋆ δ₀^{4τ})` (no `γ̄²`, audit F6). The caps: `σcap = (σK - μ/2, σB, σT, s_Lt, s_N, s_Nt, t, 0)` with the
initial values `m₀ = (σK - μ, σB - μ, σT - μ, rd.sLt, rd.sN, rd.sNt, t, 0)` and the internal caps `sB, sT` chosen
freely (`sB ≥ σB - μ`, `sT ≥ σT - μ`). -/
theorem g1_l1A {n : ℕ} (hn : 2 ≤ n) {τ : ℝ} (hτ : (n : ℝ) - 1 ≤ τ) (hτ1 : 1 ≤ τ) {ρ ρinf : ℝ}
    (hρinf : 0 < ρinf) (hρinfρ : ρinf < ρ) {M σK σB σT μ γbar : ℝ} (hM : 0 < M) (hσK : 0 < σK)
    (hσB : 0 < σB) (hσT : 0 < σT) (hμ : 0 < μ) (hγbar : 0 < γbar) (rd : RefData) (hrd : 0 < rd.M1 ∧
      0 < rd.M2 ∧ 0 < rd.sLt ∧ 0 < rd.sN ∧ 0 < rd.sNt) (d : L1Data) (hd : d.WF) (frame : Bool)
    {Mh1 Mh2 sB sT t : ℝ} (hsB : σB - μ ≤ sB) (hsT : σT - μ ≤ sT) (ht : 0 < t)
    (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ) {Xstar : ℝ} (hXpos : 0 < Xstar)
    (hX : routeAOkL1 n τ ρ ρinf rd Mh1 Mh2 (σK - μ / 2) sB sT t γbar d frame cS ctail kpre
      ![σK - μ / 2, sB, sT, rd.sLt, rd.sN, rd.sNt, t, 0]
      ![σK - μ, σB - μ, σT - μ, rd.sLt, rd.sN, rd.sNt, t, 0] μ Xstar) :
    G1BodyL1 n τ ρ ρinf M σK σB σT μ γbar rd d t cS ctail kpre
      (ρ ^ (4 * τ) / (Xstar * bite0 ρ ρinf ^ (4 * τ)))
      (ρ ^ (2 * τ) * ballSumA n τ ρ ρinf rd Mh1 Mh2 (σK - μ / 2) sB sT t γbar d frame cS ctail kpre Xstar /
        (Xstar * bite0 ρ ρinf ^ (4 * τ))) := sorry

/-- `X⋆` certifies route B with the ℓ¹ entry: as `routeBOk` of the v2 design (with all caps, audit F3, iso-F2) but with
`entryConstsB`/`entryCondB` at stage 0 and `stepConsts`/`stepCond` afterwards. `P` gives the stage inputs. -/
def routeBOkL1 (P : ℕ → StepIn) (l : EntryNumB) (Pt : StepIn) (kpre : ℕ) (σcap m₀ : Fin 8 → ℝ) (β₀ Xstar : ℝ) :
    Prop :=
  let k : ℕ → StepConsts := fun s => if s = 0 then (entryConstsB (P 0) l).1 else stepConsts (P s)
  let kt := stepConsts Pt
  let X := xSeq (P 0).κ (fun s => (k s).Q) Xstar
  let rate : StepIn → StepConsts → Fin 8 → ℝ := fun p c =>
    ![(p.n : ℝ) * c.CDK, c.CBb, p.sT ^ 2 * c.CDT / (1 - p.θ), 2 * (p.n : ℝ) * c.CDK, c.CNb, c.CNbT, c.CDT, p.Θ * c.C0 *
      (p.γb * p.δ ^ (p.κ + 1))]
  1 ≤ kpre ∧ entryCondB (P 0) l (X 0) ∧ (∀ s, 1 ≤ s → s < kpre → stepCond (P s) (X s)) ∧
    (∀ x, 0 ≤ x → x ≤ X kpre → stepCond Pt x) ∧ 2 ^ ((P 0).κ + 1) * kt.Q * X kpre ≤ 1 / 2 ∧
    (∀ i, m₀ i + (∑ s ∈ Finset.range kpre, rate (P s) (k s) i * X s) + 2 * rate Pt kt i * X kpre ≤ σcap i) ∧
    (∑ s ∈ Finset.range kpre, (P s).δ * (k s).CDK * X s) + 2 * Pt.δ * kt.CDK * X kpre < β₀

/-- Blueprint `bp:p2s2l1:g1B` (route B with the entry; it pays little: +0.1 orders on R1). `C₁ = ρ^{4τ} γ̄² s_Lt/(X⋆
δ₀^{2τ+1})`, closeness `ρ^{2τ} s_Lt S/(X⋆ δ₀^{2τ+1})` (no `γ̄²`). The regime inequality `hregime` makes the stage-0
weight `λ₀ ≤ 1` and `μ₀ = s_Lt e/(γ δ₀^τ)` for every `γ ≤ γ̄` (audit AUDIT-SHARP-V2-iso-1, F4). -/
theorem g1_l1B {n : ℕ} (hn : 2 ≤ n) {τ : ℝ} (hτ : (n : ℝ) - 1 ≤ τ) (hτ1 : 1 ≤ τ) {ρ ρinf : ℝ}
    (hρinf : 0 < ρinf) (hρinfρ : ρinf < ρ) {M σK σB σT μ γbar : ℝ} (hM : 0 < M) (hσK : 0 < σK)
    (hσB : 0 < σB) (hσT : 0 < σT) (hμ : 0 < μ) (hγbar : 0 < γbar) (rd : RefData) (hrd : 0 < rd.M1 ∧
      0 < rd.M2 ∧ 0 < rd.sLt ∧ 0 < rd.sN ∧ 0 < rd.sNt) (d : L1Data) (hd : d.WF) {t : ℝ} (ht : 0 < t)
    (P : ℕ → StepIn) (l : EntryNumB) (Pt : StepIn) (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ)
    (hP : ∀ s, (P s).n = n ∧ (P s).κ = τ ∧ (P s).δ = biteS ρ ρinf s ∧ (P s).r = stripS ρ ρinf s ∧
      (P s).c = (cS s).1 ∧ (P s).c1 = 2 ^ (τ + 1) * (cS s).2.1 ∧ (P s).cbar = (cS s).2.2 ∧ (P s).M1 = rd.M1 ∧
      (P s).M2 = rd.M2 ∧ (P s).sLt = rd.sLt ∧ (P s).sN = rd.sN ∧ (P s).sNt = rd.sNt ∧ (P s).t = t ∧
      (P s).γb = γbar ∧ (P s).sK = σK - μ / 2 ∧ σB - μ ≤ (P s).sB ∧ σT - μ ≤ (P s).sT ∧
      (P s).lam = (if s = 0 then rd.sNt * γbar * bite0 ρ ρinf ^ τ / rd.sLt else 1))
    (hregime : rd.sNt * γbar * bite0 ρ ρinf ^ τ ≤ rd.sLt)
    (hPt : Pt = { P kpre with c := ctail, c1 := 2 ^ (τ + 1) * ctail, cbar := ctail, lam := 1 })
    (hl : l = entryNumB d τ γbar γbar ρ (bite0 ρ ρinf) rd.sLt rd.sNt (rd.sLt / (γbar * bite0 ρ ρinf ^ τ))
      l.cTh l.cDTh l.tslack l.derivedT)
    {Xstar : ℝ} (hXpos : 0 < Xstar)
    (hX : routeBOkL1 P l Pt kpre ![σK - μ / 2, (P 0).sB, (P 0).sT, rd.sLt, rd.sN, rd.sNt, t, 0]
      ![σK - μ, σB - μ, σT - μ, rd.sLt, rd.sN, rd.sNt, t, 0] μ Xstar) :
    G1BodyL1 n τ ρ ρinf M σK σB σT μ γbar rd d t cS ctail kpre
      (ρ ^ (4 * τ) * γbar ^ 2 * rd.sLt / (Xstar * bite0 ρ ρinf ^ (2 * τ + 1)))
      (ρ ^ (2 * τ) * rd.sLt *
        ((∑ s ∈ Finset.range kpre, (P s).δ * (if s = 0 then (entryConstsB (P 0) l).1 else stepConsts (P s)).CDK *
          xSeq τ (fun s => (if s = 0 then (entryConstsB (P 0) l).1 else stepConsts (P s)).Q) Xstar s) +
         2 * Pt.δ * (stepConsts Pt).CDK *
          xSeq τ (fun s => (if s = 0 then (entryConstsB (P 0) l).1 else stepConsts (P s)).Q) Xstar kpre) /
        (Xstar * bite0 ρ ρinf ^ (2 * τ + 1))) := sorry

/-- The hypotheses of the frozen I-G1 body (verbatim split of `KAM.Iso.Charts.IG1Body`). -/
def IG1Hyps (n : ℕ) (τ ρ M σK σB σT μ γbar σω ν : ℝ) (ωstar : Fin n → ℝ) (D : Set ((Fin n → ℂ) × (Fin n → ℂ)))
    (H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (s γ e E₀ : ℝ) : Prop :=
  IsOpen D ∧ (∀ z ∈ D, ∀ k : Fin n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ D) ∧
    DifferentiableOn ℂ H D ∧ (∀ z ∈ D, ∀ k : Fin n → ℤ, H (z.1 + (fun i => (k i : ℂ)), z.2) = H z) ∧
    (∀ θ I : Fin n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ D →
      (H ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) ∧
    (∀ z ∈ D, ‖fderiv ℂ H z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ H z‖ ≤ M ∧
      ‖fderiv ℂ (hamiltonianVectorFieldℂ H) z‖ ≤ M ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ H) z‖ ≤ M ∧ ‖iteratedFDeriv ℂ 2 H z‖ ≤ M) ∧
    DifferentiableOn ℂ K (complexStrip n ρ) ∧ IsPrimaryOnStrip K ρ ∧ IsRealOnReals K ∧
    (∀ θ ∈ complexStrip n ρ, Metric.ball (K θ) μ ⊆ D) ∧
    (∀ θ ∈ complexStrip n ρ, ‖fderiv ℂ K θ‖ ≤ σK - μ) ∧
    (∀ θ ∈ complexStrip n ρ, IsUnit (gramMatrix K θ).det ∧
      ∀ i, ∑ j, ‖(gramMatrix K θ)⁻¹ i j‖ ≤ σB - μ) ∧
    IsUnit (KAM.Iso.extendedTorsionAverage H K (s • ωstar)).det ∧
    (∀ i, ∑ j, ‖(KAM.Iso.extendedTorsionAverage H K (s • ωstar))⁻¹ i j‖ ≤ σT - μ) ∧
    1 < s ∧ s < σω ∧ ν ≤ (s - 1) * ‖ωstar‖ ∧ ν ≤ (σω - s) * ‖ωstar‖ ∧
    0 < γ ∧ γ ≤ γbar ∧ IsDiophantine γ τ ωstar ∧ 0 ≤ e ∧
    (∀ θ ∈ complexStrip n ρ, ‖invarianceError H K (s • ωstar) θ‖ ≤ e) ∧
    ‖KAM.Iso.energyAverage H K - (E₀ : ℂ)‖ ≤ e

/-- The conclusions of the frozen I-G1 body (verbatim split of `KAM.Iso.Charts.IG1Body`). -/
def IG1Concl (n : ℕ) (τ ρ ρinf σK μ σω : ℝ) (ωstar : Fin n → ℝ) (D : Set ((Fin n → ℂ) × (Fin n → ℂ)))
    (H : (Fin n → ℂ) × (Fin n → ℂ) → ℂ) (K : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (s γ e E₀ C₂ C₃ : ℝ) :
    Prop :=
  ∃ (Kinf : (Fin n → ℂ) → (Fin n → ℂ) × (Fin n → ℂ)) (sinf : ℝ),
    1 < sinf ∧ sinf < σω ∧ DifferentiableOn ℂ Kinf (complexStrip n ρinf) ∧
    IsPrimaryOnStrip Kinf ρinf ∧ IsRealOnReals Kinf ∧
    (∀ θ ∈ complexStrip n ρinf, invarianceError H Kinf (sinf • ωstar) θ = 0) ∧
    (∀ θ ∈ complexStrip n ρinf, H (Kinf θ) = (E₀ : ℂ)) ∧
    (∀ θ ∈ complexStrip n ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ D) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
    (∀ θ ∈ complexStrip n ρinf, ‖Kinf θ - K θ‖ ≤ C₂ * e / (γ ^ 2 * ρ ^ (2 * τ))) ∧
    ‖sinf • ωstar - s • ωstar‖ ≤ C₃ * e / (γ ^ 2 * ρ ^ (2 * τ))

/-- The split is the frozen body (definitional). -/
theorem ig1Body_iff_split (n : ℕ) (τ ρ ρinf M σK σB σT μ γbar σω ν : ℝ) (ωstar : Fin n → ℝ) (C₁ C₂ C₃ : ℝ) :
    KAM.Iso.Charts.IG1Body n τ ρ ρinf M σK σB σT μ γbar σω ν ωstar C₁ C₂ C₃ ↔
      ∀ D H K s γ e E₀, IG1Hyps n τ ρ M σK σB σT μ γbar σω ν ωstar D H K s γ e E₀ →
        C₁ * e / (γ ^ 4 * ρ ^ (4 * τ)) < 1 → IG1Concl n τ ρ ρinf σK μ σω ωstar D H K s γ e E₀ C₂ C₃ := by
  constructor
  · intro HB D H K s γ e E₀ ⟨h1, h2, h3, h4, h5, h6, h7, h8, h9, h10, h11, h12, h13, h14, h15, h16, h17, h18, h19, h20, h21, h22, h23, h24⟩ hs
    exact HB D H K s γ e E₀ h1 h2 h3 h4 h5 h6 h7 h8 h9 h10 h11 h12 h13 h14 h15 h16 h17 h18 h19 h20 h21 h22 h23 h24 hs
  · intro HB D H K s γ e E₀ h1 h2 h3 h4 h5 h6 h7 h8 h9 h10 h11 h12 h13 h14 h15 h16 h17 h18 h19 h20 h21 h22 h23 h24 hs
    exact HB D H K s γ e E₀ ⟨h1, h2, h3, h4, h5, h6, h7, h8, h9, h10, h11, h12, h13, h14, h15, h16, h17, h18, h19, h20, h21, h22, h23, h24⟩ hs

/-- The isoenergetic body with the ℓ¹ entry: the frozen hypotheses, the refined data, the torsion cap, the ℓ¹ entry data
of `K` at the initial frequency `s • ω*` (`Γ_∞` premises at `s • ω*`; a certificate at `ω*` serves by
`smallDivisorSup_smul_le`), and the sup small-divisor premises on the ray (L5), imply the frozen conclusion. -/
def IG1BodyL1 (n : ℕ) (τ ρ ρinf M σK σB σT μ γbar σω ν : ℝ) (ωstar : Fin n → ℝ) (rd : RefData) (d : L1Data)
    (t : ℝ) (cS : ℕ → ℝ × ℝ × ℝ) (ctail : ℝ) (kpre : ℕ) (C₁ C₂ C₃ : ℝ) : Prop :=
  ∀ D H K s γ e E₀, IG1Hyps n τ ρ M σK σB σT μ γbar σω ν ωstar D H K s γ e E₀ →
    RefHyp rd D H K ρ → (∀ θ ∈ complexStrip n (ρ - bite0 ρ ρinf), ‖torsion H K (s • ωstar) θ‖ ≤ t) →
    L1Hyp d H K (s • ωstar) ρ (bite0 ρ ρinf) e rd.sLt rd.sNt →
    (∀ j < kpre, SDConst n τ γ (biteS ρ ρinf j) (cS j).1 (ray ωstar) ∧
      SDConst n τ γ (biteS ρ ρinf j / 2) (cS j).2.1 (ray ωstar) ∧
      SDConst n τ γ (stripS ρ ρinf j) (cS j).2.2 (ray ωstar)) →
    (∀ δ, 0 < δ → δ ≤ biteS ρ ρinf kpre → SDConst n τ γ δ ctail (ray ωstar)) →
    C₁ * e / (γ ^ 4 * ρ ^ (4 * τ)) < 1 → IG1Concl n τ ρ ρinf σK μ σω ωstar D H K s γ e E₀ C₂ C₃

/-- Blueprint `bpI:p2s2l1:g1` (route A; route B analogous): `C₁ = ρ^{4τ}/(X⋆ δ₀^{4τ})`, `C₂ = ρ^{2τ} S̃/(X⋆ δ₀^{4τ})`,
`C₃` the frequency sum (entry included) normalised the same way. The smallness predicate is `routeAOkL1` with the
isoenergetic stage inputs (`iso := true`, `Θ = σω ‖ω*‖`). -/
