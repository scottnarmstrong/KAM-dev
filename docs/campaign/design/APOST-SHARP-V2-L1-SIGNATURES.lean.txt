/-- The optional ℓ¹ frame certificate: caps of `‖L‖_{1,ρ}`, `‖N‖_{1,ρ}` (AUDIT-B10D-1, F1). -/
structure FrameL1 where
  nuK : ℝ
  nuN : ℝ

/-- The ℓ¹ entry numbers of route A: `‖ηᴺ‖_{1,ρ} ≤ aN e`, `‖ηᴸ‖_{1,ρ} ≤ aL e`, `‖T‖_{1,ρ-δ₀} ≤ t1`,
`sup_ℝ |T| ≤ tR`, `Γ_∞(ω, δ₀) ≤ G0`, `Γ_∞(ω, ρ) ≤ Gρ`, and the optional frame certificate. -/
structure EntryA where
  aN : ℝ
  aL : ℝ
  t1 : ℝ
  tR : ℝ
  G0 : ℝ
  Gρ : ℝ
  frame : Option FrameL1 := none

/-- Well-formedness of the entry numbers; an enabled frame option has positive caps (F1). -/
structure EntryA.Wf (E : EntryA) : Prop where
  aN : 0 ≤ E.aN
  aL : 0 ≤ E.aL
  t1 : 0 ≤ E.t1
  tR : 0 ≤ E.tR
  G0 : 0 ≤ E.G0
  Gρ : 0 ≤ E.Gρ
  frame : ∀ F, E.frame = some F → 0 < F.nuK ∧ 0 < F.nuN

/-- The entry hypotheses on explicit objects of `(h, K, ω)` at the frozen sup bound `e` (a certificate
checks them); the frame bounds are required exactly when the option is enabled (F1). -/
structure EntryHypA (N : NumA) (E : EntryA) (h : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ)
    (K : (Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ)) (ω : Fin N.n → ℝ) (e : ℝ) : Prop where
  l1EtaN : L1Vec N.ρ (etaN h K ω) (E.aN * e)
  l1EtaL : L1Vec N.ρ (etaL h K ω) (E.aL * e)
  l1Torsion : L1Mat (N.ρ - N.δ 0) (torsion h K ω) E.t1
  realTorsion : ∀ x : Fin N.n → ℝ, ‖torsion h K ω (fun i => (x i : ℂ))‖ ≤ E.tR
  gam0 : smallDivisorSup ω (N.δ 0) ≤ E.G0
  gamρ : smallDivisorSup ω N.ρ ≤ E.Gρ
  frame : ∀ F, E.frame = some F → L1Mat N.ρ (tangentMat K) F.nuK ∧ L1Mat N.ρ (normalMat K) F.nuN

/-- The constants of the entry stage. -/
structure EntryConstsA where
  CS : ℝ
  CXiN : ℝ
  CXiN1 : ℝ
  CG : ℝ
  CXiL : ℝ
  CXiL1 : ℝ
  CDK : ℝ
  COmK : ℝ
  CE : ℝ
  dL : ℝ
  dLt : ℝ
  rB : ℝ
  rN : ℝ
  rNt : ℝ
  CDT : ℝ
  rT : ℝ
  CDw : ℝ
  gram : ℝ

/-- Blueprint `bp:p2s2l1:constA`, rebased on B9's `StepInA` (stage `0` inputs `N.inp 0`): each ingredient
is the smaller of its sup bound (B9's formula) and its ℓ¹ bound; the frame increments are products, with
no division (F1); the average uses `min t tR` (F6). For `E.frame = none` and the sup branches of every
minimum the constants equal B9's (`rateV`, `Cond`) as real numbers (by `ring`, not definitionally). -/
def entryConsts (p : StepInA) (E : EntryA) : EntryConstsA :=
  let n : ℝ := p.n
  let einv := (Real.exp 1)⁻¹
  let tavg := min p.t E.tR
  let cavg := min (p.ca * p.sLt) (E.Gρ * E.aN * p.g)
  let CS := if p.iso then p.sT * max (p.sNt * p.g + tavg * cavg) (p.g + cavg * p.HY)
    else p.sT * (p.sNt * p.g + tavg * cavg)
  let cN1 := E.G0 * E.aN * p.g
  let CXiN := CS + min (p.c * p.sLt) cN1
  let CXiN1 := CS + cN1
  let CG := p.t * CXiN + p.Θ * CS + p.sNt * p.g
  let CXiL1 := E.G0 * p.g * (E.t1 * CXiN1 + p.Θ * CS + E.aL * p.g)
  let CXiL := min (p.c * CG) CXiL1
  let CXi := max CXiL (p.g * CXiN)
  let CDK := p.sK * CXiL + p.sN * p.g * CXiN
  let COmK := min p.COm (n * einv * E.G0 * E.aN * p.g)
  let CSym := max 1 (p.sB ^ 2) * COmK
  let CDXi := max CG (p.sLt * p.g)
  let CLin := p.CRed * CXi + CSym * CDXi + p.Θ * COmK * CS
  let CdK := p.CFr * CXi + p.P * p.g * CDXi
  let CE1 := 2 * p.P * (p.γb ^ 2 * p.δ ^ (2 * p.κ - 1)) * CLin + p.M2 / 2 * CDK ^ 2 +
    (if p.iso then p.g * CS * CdK else 0)
  let CE2 := p.γb * p.δ ^ (p.κ - 1) * n * (p.sLt * p.g + p.Θ * COmK) * CXiL + p.Mh2 / 2 * CDK ^ 2
  let CE := if p.iso then max CE1 CE2 else CE1
  let dL := match E.frame with
    | some F => min (n * CDK) (einv * (F.nuK * CXiL1 + F.nuN * p.g * CXiN1))
    | none => n * CDK
  let dLt := match E.frame with
    | some F => min (2 * n * CDK) (2 * n * einv * (F.nuK * CXiL1 + F.nuN * p.g * CXiN1))
    | none => 2 * n * CDK
  let rB := p.sB ^ 2 * (p.sLt * dL + p.sK * dLt) / (1 - p.th)
  let rN := p.sB * dL + p.sK * rB
  let rNt := rB * p.sLt + p.sB * dLt
  let CDw := p.Θ * p.g * CS
  let CDT0 := 2 * p.M1 * (rNt * p.sN + p.sNt * rN) + 2 * p.M2 * p.sNt * p.sN * p.δ * CDK +
    p.sB ^ 3 * p.w * (p.sK * dLt + p.sLt * dL) + 2 * n * p.sNt * p.sB * (p.γb ^ 2 * p.δ ^ (2 * p.κ)) +
    p.sB ^ 3 * p.w * (p.COm + COmK) * p.g
  let CDT := if p.iso then
      max (CDT0 + CDw * p.δ) (n * CDw * p.δ + 2 * n * p.sNt * (p.γb ^ 2 * p.δ ^ (2 * p.κ)))
    else CDT0
  ⟨CS, CXiN, CXiN1, CG, CXiL, CXiL1, CDK, COmK, CE, dL, dLt, rB, rN, rNt, CDT,
    p.sT ^ 2 / (1 - p.th) * CDT, CDw, p.sB * (p.sLt * dL + p.sK * dLt)⟩

/-- The entry step conditions at `x = ε/(γ⁴ δ₀^{2a})` (B9's `StepInA.Cond` with the entry constants). -/
structure EntryCond (p : StepInA) (E : EntryA) (x : ℝ) : Prop where
  err : p.γb ^ 4 * p.δ ^ (2 * (2 * p.κ) - 1) * x ≤ 1
  sym : max 1 (p.sB ^ 2) * (entryConsts p E).COmK * (p.γb ^ 3 * p.δ ^ (3 * p.κ - 1)) * x ≤ 1 / 2
  quad : (entryConsts p E).CE * x ≤ 1
  twist : p.sT * (entryConsts p E).CDT * (p.γb ^ 2 * p.δ ^ (2 * p.κ - 1)) * x ≤ p.th
  gram : (entryConsts p E).gram * (p.γb ^ 2 * p.δ ^ (2 * p.κ - 1)) * x ≤ p.th

/-- The entry stage (fixed frequency; B9's rate order: ball, DK, B, ⟨T⟩⁻¹, Lᵀ, N, Nᵀ, T). -/
def entryStageA (p : StepInA) (E : EntryA) : StageA 8 where
  CE := (entryConsts p E).CE
  CD := (entryConsts p E).CDK
  Cp := 0
  rate := ![(entryConsts p E).CDK, (entryConsts p E).dL, (entryConsts p E).rB, (entryConsts p E).rT,
    (entryConsts p E).dLt, (entryConsts p E).rN, (entryConsts p E).rNt, (entryConsts p E).CDT]

/-- The stages of a run with the ℓ¹ entry: stage `0` is the entry, every later stage is B9's. -/
def stageL1 (N : NumA) (E : EntryA) (s : ℕ) : StageA 8 :=
  if s = 0 then entryStageA (N.inp 0) E else N.stage s

/-- The conditions of a run with the ℓ¹ entry. -/
def condL1 (N : NumA) (E : EntryA) (s : ℕ) (x : ℝ) : Prop :=
  if s = 0 then EntryCond (N.inp 0) E x else (N.inp s).Cond x

/-- The normalised errors of a run with the ℓ¹ entry. -/
def xSL1 (N : NumA) (E : EntryA) (Xs : ℝ) : ℕ → ℝ := xA (2 * N.τ) (fun s => (stageL1 N E s).CE) Xs

/-- The route-A scheme with the ℓ¹ entry: B9's `schemeA`, whose stage-`0` admissibility also carries the
entry hypotheses at the current error. The schedule is B9's (`deltaSeq 3`), unchanged (F5). -/
def schemeAL1 (N : NumA) (E : EntryA) (𝔅 : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ)))
    (h : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ) (ω : Fin N.n → ℝ) (μ : ℝ) :
    SchemeA N.n 8 ((Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ)) where
  tor := fun K => K
  par := fun _ => 0
  Adm := fun s ε m K => (schemeA N 𝔅 h ω μ).Adm s ε m K ∧ (s = 0 → EntryHypA N E h K ω ε)
  step := (schemeA N 𝔅 h ω μ).step

/-- Blueprint `bp:p2s2l1:entryA` in the engine: (N1), (N2) of the scheme with the ℓ¹ entry. Stage `0` is
the entry step (the B9 step with the ℓ¹ ingredients of B10a/B10b), every later stage is
`KAM.Sharp2.RouteA.schemeA_isNewton`. -/
theorem schemeAL1_isNewton {N : NumA} {E : EntryA} (hN : N.Wf) (hE : E.Wf) (hfix : N.iso = false)
    (hΘ : N.Θ = 0) {τ M γ : ℝ} {𝔅 : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ))}
    {h : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ} {ω : Fin N.n → ℝ} {Ω : Set (Fin N.n → ℝ)}
    (hS : Standing N.n τ M N.sK N.sB N.sT 𝔅 h ω γ)
    (hd1 : ∀ z ∈ 𝔅, ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ N.M1)
    (hd2 : ∀ z ∈ 𝔅, ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ N.M2)
    (hγb : γ ≤ N.γb) (hω : ω ∈ Ω) (hC : N.Cert γ Ω) {μ b : ℝ} (hb : b < μ) :
    (schemeAL1 N E 𝔅 h ω μ).IsNewton (2 * N.τ) 3 N.ρ N.ρinf γ (stageL1 N E) jA (condL1 N E)
      (N.capV b) := sorry

/-- (J1)–(J3) for route A with the ℓ¹ entry at `X⋆`: the entry conditions at `X⋆`, B9's conditions from
stage `1`, the tail, and every cap sum with initial values `m₀` and caps `σ` (F2). -/
structure routeAOkL1 (N : NumA) (E : EntryA) (σ m₀ : Fin 8 → ℝ) (Xs : ℝ) : Prop where
  kp_pos : 1 ≤ N.kp
  Xs_nonneg : 0 ≤ Xs
  J0 : EntryCond (N.inp 0) E Xs
  J1 : ∀ s, 1 ≤ s → s < N.kp → (N.inp s).Cond (xSL1 N E Xs s)
  J1t : (N.inp N.kp).Cond (xSL1 N E Xs N.kp)
  J2 : 2 ^ (2 * (2 * N.τ)) * (N.stage N.kp).CE * xSL1 N E Xs N.kp ≤ 1
  J3 : ∀ i, m₀ i + sumA (2 * N.τ) 3 N.ρ N.ρinf N.γb (xSL1 N E Xs) (fun s => (stageL1 N E s).rate i)
    (2 * N.τ + (jA i : ℝ)) N.kp ≤ σ i

/-- `routeAOkL1` is the engine's `RunData` (nonnegativity of the entry constants from `NumA.Wf`,
`EntryA.Wf`; downward closure of `EntryCond`; the tail is B9's since `1 ≤ kp`). -/
theorem routeAOkL1.runData {N : NumA} {E : EntryA} (hN : N.Wf) (hE : E.Wf) {σ m₀ : Fin 8 → ℝ}
    {Xs : ℝ} (h : routeAOkL1 N E σ m₀ Xs) :
    RunData (2 * N.τ) 3 N.ρ N.ρinf N.γb (stageL1 N E) jA (condL1 N E) σ m₀ N.kp Xs := sorry

/-- `C₂` of the entry branch (the ball sum, no `γ̄²`). -/
def c2AL1 (N : NumA) (E : EntryA) (Xs : ℝ) : ℝ :=
  N.ρ ^ (2 * N.τ) * sumA (2 * N.τ) 3 N.ρ N.ρinf 1 (xSL1 N E Xs) (fun s => (stageL1 N E s).CD)
      (2 * N.τ) N.kp /
    (Xs * delta0 3 N.ρ N.ρinf ^ (4 * N.τ))

/-- The frozen G1 body with B9's refined data and certificates and the ℓ¹ entry hypotheses. -/
def G1BodyAL1 (N : NumA) (I : InitA) (E : EntryA) (M σK σB σT μ γbar C₁ C₂ : ℝ) : Prop :=
  ∀ (𝔅 : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ))) (h : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ)
    (K : (Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ)) (ω : Fin N.n → ℝ) (γ e : ℝ)
    (Ω : Set (Fin N.n → ℝ)),
    IsOpen 𝔅 →
    (∀ z ∈ 𝔅, ∀ k : Fin N.n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ 𝔅) →
    DifferentiableOn ℂ h 𝔅 →
    (∀ z ∈ 𝔅, ∀ k : Fin N.n → ℤ, h (z.1 + (fun i => (k i : ℂ)), z.2) = h z) →
    (∀ θ I : Fin N.n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ 𝔅 →
      (h ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) →
    (∀ z ∈ 𝔅, ‖fderiv ℂ h z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ h z‖ ≤ M ∧
      ‖fderiv ℂ (hamiltonianVectorFieldℂ h) z‖ ≤ M ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ h) z‖ ≤ M) →
    DifferentiableOn ℂ K (complexStrip N.n N.ρ) → IsPrimaryOnStrip K N.ρ → IsRealOnReals K →
    (∀ θ ∈ complexStrip N.n N.ρ, Metric.ball (K θ) μ ⊆ 𝔅) →
    (∀ θ ∈ complexStrip N.n N.ρ, ‖fderiv ℂ K θ‖ ≤ σK - μ) →
    (∀ θ ∈ complexStrip N.n N.ρ, IsUnit (gramMatrix K θ).det ∧
      ∀ i, ∑ j, ‖(gramMatrix K θ)⁻¹ i j‖ ≤ σB - μ) →
    IsUnit (torsionAverage h K ω).det →
    (∀ i, ∑ j, ‖(torsionAverage h K ω)⁻¹ i j‖ ≤ σT - μ) →
    RefHypA N I 𝔅 h K ω → ω ∈ Ω → N.Cert γ Ω → EntryHypA N E h K ω e →
    0 < γ → γ ≤ γbar → IsDiophantine γ N.τ ω →
    (∀ θ ∈ complexStrip N.n N.ρ, ‖invarianceError h K ω θ‖ ≤ e) →
    C₁ * e / (γ ^ 4 * N.ρ ^ (4 * N.τ)) < 1 →
    ∃ Kinf : (Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ),
      DifferentiableOn ℂ Kinf (complexStrip N.n N.ρinf) ∧ IsPrimaryOnStrip Kinf N.ρinf ∧
      IsRealOnReals Kinf ∧
      (∀ θ ∈ complexStrip N.n N.ρinf, invarianceError h Kinf ω θ = 0) ∧
      (∀ θ ∈ complexStrip N.n N.ρinf, Metric.ball (Kinf θ) (μ / 2) ⊆ 𝔅) ∧
      (∀ θ ∈ complexStrip N.n N.ρinf, ‖fderiv ℂ Kinf θ‖ ≤ σK - μ / 2) ∧
      ∀ θ ∈ complexStrip N.n N.ρinf, ‖Kinf θ - K θ‖ ≤ C₂ * e / (γ ^ 2 * N.ρ ^ (2 * N.τ))

/-- Blueprint `bp:p2s2l1:g1A` / `bp:p2s2l1:frozen` (route A, the PlasmaLean route), repaired: three
branches — v1 (`sharpC1`), B9's sup route A at `X⋆_A` (`routeAOk`), the ℓ¹ entry at `X⋆` (`routeAOkL1`)
— with caps `capV (μ/2)` and initial values `initV I` (F2); `C₁` the minimum, `C₂` the maximum, proved by
cases on the branch that applies (F6). Never worse than v1 or than B9's route A by construction. -/
theorem g1_v2A_l1 {N : NumA} {E : EntryA} (hN : N.Wf) (hE : E.Wf) (hfix : N.iso = false)
    (hΘ : N.Θ = 0) (hτ : (N.n : ℝ) - 1 ≤ N.τ) {M σK σB σT μ γbar XsA Xs : ℝ} (I : InitA) (hM : 0 < M)
    (hσK : 0 < σK) (hσB : 0 < σB) (hσT : 0 < σT) (hμ : 0 < μ) (hγbar : γbar = N.γb)
    (hsK : N.sK = σK - μ / 2)
    (hOkA : routeAOk N (N.capV (μ / 2)) (initV I σK σB σT μ) XsA) (hXApos : 0 < XsA)
    (hOk : routeAOkL1 N E (N.capV (μ / 2)) (initV I σK σB σT μ) Xs) (hXpos : 0 < Xs) :
    0 < min (sharpC1 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar μ N.τ N.ρ N.ρinf)
        (min (c1A N XsA) (c1A N Xs)) ∧
      0 < max (sharpC2 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar N.τ N.ρ N.ρinf)
        (max (c2A N XsA) (c2AL1 N E Xs)) ∧
      G1BodyAL1 N I E M σK σB σT μ γbar
        (min (sharpC1 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar μ N.τ N.ρ N.ρinf)
          (min (c1A N XsA) (c1A N Xs)))
        (max (sharpC2 N.n N.τ (cohConstR N.n N.τ) M σK σB σT γbar N.τ N.ρ N.ρinf)
          (max (c2A N XsA) (c2AL1 N E Xs))) := sorry

/-- The isoenergetic entry stage: B9's nine rates (the last one the frequency rate `C_Δω`) with the entry
constants; `Cp = C_Δω`. -/
def entryStageIA (p : StepInA) (E : EntryA) : StageA 9 where
  CE := (entryConsts p E).CE
  CD := (entryConsts p E).CDK
  Cp := (entryConsts p E).CDw
  rate := ![(entryConsts p E).CDK, (entryConsts p E).dL, (entryConsts p E).rB, (entryConsts p E).rT,
    (entryConsts p E).dLt, (entryConsts p E).rN, (entryConsts p E).rNt, (entryConsts p E).CDT,
    (entryConsts p E).CDw]

/-- The stages of an isoenergetic run with the ℓ¹ entry. -/
def stageIL1 (N : NumA) (E : EntryA) (s : ℕ) : StageA 9 :=
  if s = 0 then entryStageIA (N.inp 0) E else stageI N s

/-- The normalised errors of an isoenergetic run with the ℓ¹ entry. -/
def xIL1 (N : NumA) (E : EntryA) (Xs : ℝ) : ℕ → ℝ := xA (2 * N.τ) (fun s => (stageIL1 N E s).CE) Xs

/-- (J1)–(J3) of the isoenergetic route A with the ℓ¹ entry (caps `capI`, initial values `initI`). -/
structure routeAOkIL1 (N : NumA) (E : EntryA) (σ m₀ : Fin 9 → ℝ) (Xs : ℝ) : Prop where
  kp_pos : 1 ≤ N.kp
  Xs_nonneg : 0 ≤ Xs
  J0 : EntryCond (N.inp 0) E Xs
  J1 : ∀ s, 1 ≤ s → s < N.kp → (N.inp s).Cond (xIL1 N E Xs s)
  J1t : (N.inp N.kp).Cond (xIL1 N E Xs N.kp)
  J2 : 2 ^ (2 * (2 * N.τ)) * (stageI N N.kp).CE * xIL1 N E Xs N.kp ≤ 1
  J3 : ∀ i, m₀ i + sumA (2 * N.τ) 3 N.ρ N.ρinf N.γb (xIL1 N E Xs) (fun s => (stageIL1 N E s).rate i)
    (2 * N.τ + (jI i : ℝ)) N.kp ≤ σ i

/-- `routeAOkIL1` is the engine's `RunData` (as `routeAOkL1.runData`, with B9's `routeAOkI.runData` from
stage `1`). -/
theorem routeAOkIL1.runData {N : NumA} {E : EntryA} (hN : N.Wf) (hE : E.Wf) {σ m₀ : Fin 9 → ℝ}
    {Xs : ℝ} (h : routeAOkIL1 N E σ m₀ Xs) :
    RunData (2 * N.τ) 3 N.ρ N.ρinf N.γb (stageIL1 N E) jI (condL1 N E) σ m₀ N.kp Xs := sorry

/-- `C₂` of the isoenergetic entry branch (ball sum). -/
def c2IL1 (N : NumA) (E : EntryA) (Xs : ℝ) : ℝ :=
  N.ρ ^ (2 * N.τ) * sumA (2 * N.τ) 3 N.ρ N.ρinf 1 (xIL1 N E Xs) (fun s => (stageIL1 N E s).CD)
      (2 * N.τ) N.kp /
    (Xs * delta0 3 N.ρ N.ρinf ^ (4 * N.τ))

/-- `C₃` of the isoenergetic entry branch: the normalised complete frequency sum, entry included (F3). -/
def c3IL1 (N : NumA) (E : EntryA) (Xs : ℝ) : ℝ :=
  N.ρ ^ (2 * N.τ) * sumA (2 * N.τ) 3 N.ρ N.ρinf 1 (xIL1 N E Xs) (fun s => (stageIL1 N E s).Cp)
      (2 * N.τ) N.kp /
    (Xs * delta0 3 N.ρ N.ρinf ^ (4 * N.τ))

/-- B9's isoenergetic scheme whose stage-`0` admissibility carries the entry hypotheses at the initial
frequency `s ω*`. -/
def schemeIAL1 (N : NumA) (E : EntryA) (D : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ)))
    (H : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ) (ωstar : Fin N.n → ℝ) (E₀ σω ν μ : ℝ) :
    SchemeA N.n 9 (IsoState N.n) where
  tor := fun x => x.1
  par := fun x => x.2 * ‖ωstar‖
  Adm := fun s ε m x => (schemeIA N D H ωstar E₀ σω ν μ).Adm s ε m x ∧
    (s = 0 → EntryHypA N E H x.1 (x.2 • ωstar) ε)
  step := (schemeIA N D H ωstar E₀ σω ν μ).step

/-- (N1), (N2) of the isoenergetic scheme with the ℓ¹ entry; the iso inputs are B9's: `iso = true`,

theorem schemeIAL1_isNewton {N : NumA} {E : EntryA} (hN : N.Wf) (hE : E.Wf) (hiso : N.iso = true)
    {τ M σω ν γ E₀ : ℝ} {D : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ))}

    (hΘ : N.Θ = σω * ‖ωstar‖) (hS : IsoStanding N.n τ M N.sK N.sB N.sT σω ν ωstar D H γ)
    (hd1 : ∀ z ∈ D, ‖fderiv ℂ (hamiltonianVectorFieldℂ H) z‖ ≤ N.M1)
    (hd2 : ∀ z ∈ D, ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ H) z‖ ≤ N.M2)

    (hb : b < μ) (hbν : bν < ν) :

/-- The isoenergetic frozen body with B9's refined data and certificates and the ℓ¹ entry hypotheses at
the initial frequency. -/
def IG1BodyAL1 (N : NumA) (I : InitA) (E : EntryA) (Mh1 M σK σB σT μ γbar σω ν : ℝ)
    (ωstar : Fin N.n → ℝ) (C₁ C₂ C₃ : ℝ) : Prop :=
  ∀ (D : Set ((Fin N.n → ℂ) × (Fin N.n → ℂ))) (H : (Fin N.n → ℂ) × (Fin N.n → ℂ) → ℂ)
    (K : (Fin N.n → ℂ) → (Fin N.n → ℂ) × (Fin N.n → ℂ)) (s γ e E₀ : ℝ) (Ω : Set (Fin N.n → ℝ)),
    IsOpen D → (∀ z ∈ D, ∀ k : Fin N.n → ℤ, (z.1 + (fun i => (k i : ℂ)), z.2) ∈ D) →
    DifferentiableOn ℂ H D →
    (∀ z ∈ D, ∀ k : Fin N.n → ℤ, H (z.1 + (fun i => (k i : ℂ)), z.2) = H z) →
    (∀ θ I : Fin N.n → ℝ, ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ))) ∈ D →
      (H ((fun i => (θ i : ℂ)), (fun i => (I i : ℂ)))).im = 0) →
    (∀ z ∈ D, ‖fderiv ℂ H z‖ ≤ M ∧ ‖hamiltonianVectorFieldℂ H z‖ ≤ M ∧
      ‖fderiv ℂ (hamiltonianVectorFieldℂ H) z‖ ≤ M ∧
      ‖iteratedFDeriv ℂ 2 (hamiltonianVectorFieldℂ H) z‖ ≤ M ∧ ‖iteratedFDeriv ℂ 2 H z‖ ≤ M) →
    DifferentiableOn ℂ K (complexStrip N.n N.ρ) → IsPrimaryOnStrip K N.ρ → IsRealOnReals K →
    (∀ θ ∈ complexStrip N.n N.ρ, Metric.ball (K θ) μ ⊆ D) →

    IsUnit (KAM.Iso.extendedTorsionAverage H K (s • ωstar)).det →

    1 < s → s < σω → ν ≤ (s - 1) * ‖ωstar‖ → ν ≤ (σω - s) * ‖ωstar‖ →
    RefHypIA N I Mh1 D H K (s • ωstar) → (∀ s' : ℝ, 1 ≤ s' → s' • ωstar ∈ Ω) → N.Cert γ Ω →

      (∀ θ ∈ complexStrip N.n N.ρinf, H (Kinf θ) = (E₀ : ℂ)) ∧
